   468→   468→/**
   469→   469→ * Pan + wheel-zoom canvas for the upgrade tree.
   470→   470→ * Defined at module scope (NOT inside UpgradesScreen) so it keeps a stable
   471→   471→ * component identity across re-renders — otherwise its refs/transform would
   472→   472→ * reset every time the parent updates state.
   473→   473→ */
   474→   474→function UpgradeTreeCanvas({
   475→   475→  children,
   476→   476→  onZoomChange,
   477→   477→}: {
   478→   478→  children: React.ReactNode;
   479→   479→  onZoomChange: (zoom: number) => void;
   480→   480→}) {
   481→   481→  const { t } = useLanguage();
   482→   482→  const containerRef = useRef<HTMLDivElement>(null);
   483→   483→  const contentRef = useRef<HTMLDivElement>(null);
   484→   484→  const transformRef = useRef({ zoom: 1, panX: 0, panY: 0 });
   485→   485→  // displayZoom mirrors the ref into state so we can read it during render
   486→   486→  // for the zoom indicator (refs must not be read during render).
   487→   487→  const [displayZoom, setDisplayZoom] = useState(1);
   488→   488→  const [isDragging, setIsDragging] = useState(false);
   489→   489→  const dragStart = useRef({ x: 0, y: 0, panX: 0, panY: 0, moved: false });
   490→   490→
   491→   491→  const applyTransform = useCallback(() => {
   492→   492→    const el = contentRef.current;
   493→   493→    if (el) {
   494→   494→      const { zoom, panX, panY } = transformRef.current;
   495→   495→      el.style.transform = `translate(${panX}px, ${panY}px) scale(${zoom})`;
   496→   496→    }
   497→   497→    onZoomChange(transformRef.current.zoom);
   498→   498→    setDisplayZoom(transformRef.current.zoom);
   499→   499→  }, [onZoomChange]);
   500→   500→
   501→   501→  // Center the content on first mount
   502→   502→  useEffect(() => {
   503→   503→    const el = containerRef.current;
   504→   504→    if (!el) return;
   505→   505→    const cw = el.clientWidth;
   506→   506→    const ch = el.clientHeight;
   507→   507→    transformRef.current = {
   508→   508→      zoom: 1,
   509→   509→      panX: cw / 2 - CENTER,
   510→   510→      panY: ch / 2 - CENTER,
   511→   511→    };
   512→   512→    applyTransform();
   513→   513→    const handleResize = () => {
   514→   514→      const z = transformRef.current.zoom;
   515→   515→      transformRef.current.panX = el.clientWidth / 2 - CENTER * z;
   516→   516→      transformRef.current.panY = el.clientHeight / 2 - CENTER * z;
   517→   517→      applyTransform();
   518→   518→    };
   519→   519→    window.addEventListener('resize', handleResize);
   520→   520→    return () => window.removeEventListener('resize', handleResize);
   521→   521→  }, [applyTransform]);
   522→   522→
   523→   523→  // Non-passive wheel listener so we can preventDefault page scroll
   524→   524→  useEffect(() => {
   525→   525→    const el = containerRef.current;
   526→   526→    if (!el) return;
   527→   527→    const onWheel = (e: WheelEvent) => {
   528→   528→      e.preventDefault();
   529→   529→      const rect = el.getBoundingClientRect();
   530→   530→      const cx = e.clientX - rect.left;
   531→   531→      const cy = e.clientY - rect.top;
   532→   532→      const { zoom: prevZoom, panX: prevPanX, panY: prevPanY } = transformRef.current;
   533→   533→      const factor = e.deltaY < 0 ? 1.12 : 1 / 1.12;
   534→   534→      const newZoom = clampZoom(prevZoom * factor);
   535→   535→      const actualFactor = newZoom / prevZoom;
   536→   536→      transformRef.current.zoom = newZoom;
   537→   537→      transformRef.current.panX = cx - (cx - prevPanX) * actualFactor;
   538→   538→      transformRef.current.panY = cy - (cy - prevPanY) * actualFactor;
   539→   539→      applyTransform();
   540→   540→    };
   541→   541→    el.addEventListener('wheel', onWheel, { passive: false });
   542→   542→    return () => el.removeEventListener('wheel', onWheel);
   543→   543→  }, [applyTransform]);
   544→   544→
   545→   545→  const onMouseDown = (e: React.MouseEvent) => {
   546→   546→    if (e.button !== 0) return;
   547→   547→    dragStart.current = {
   548→   548→      x: e.clientX,
   549→   549→      y: e.clientY,
   550→   550→      panX: transformRef.current.panX,
   551→   551→      panY: transformRef.current.panY,
   552→   552→      moved: false,
   553→   553→    };
   554→   554→    setIsDragging(true);
   555→   555→  };
   556→   556→
   557→   557→  const onMouseMove = (e: React.MouseEvent) => {
   558→   558→    if (!isDragging) return;
   559→   559→    const dx = e.clientX - dragStart.current.x;
   560→   560→    const dy = e.clientY - dragStart.current.y;
   561→   561→    if (Math.abs(dx) > 3 || Math.abs(dy) > 3) dragStart.current.moved = true;
   562→   562→    transformRef.current.panX = dragStart.current.panX + dx;
   563→   563→    transformRef.current.panY = dragStart.current.panY + dy;
   564→   564→    applyTransform();
   565→   565→  };
   566→   566→
   567→   567→  const zoomBy = useCallback((factor: number) => {
   568→   568→    const el = containerRef.current;
   569→   569→    if (!el) return;
   570→   570→    const cx = el.clientWidth / 2;
   571→   571→    const cy = el.clientHeight / 2;
   572→   572→    const { zoom: prevZoom, panX: prevPanX, panY: prevPanY } = transformRef.current;
   573→   573→    const newZoom = clampZoom(prevZoom * factor);
   574→   574→    const actualFactor = newZoom / prevZoom;
   575→   575→    transformRef.current.zoom = newZoom;
   576→   576→    transformRef.current.panX = cx - (cx - prevPanX) * actualFactor;
   577→   577→    transformRef.current.panY = cy - (cy - prevPanY) * actualFactor;
   578→   578→    applyTransform();
   579→   579→  }, [applyTransform]);
   580→   580→
   581→   581→  const resetView = useCallback(() => {
   582→   582→    const el = containerRef.current;
   583→   583→    if (!el) return;
   584→   584→    transformRef.current = {
   585→   585→      zoom: 1,
   586→   586→      panX: el.clientWidth / 2 - CENTER,
   587→   587→      panY: el.clientHeight / 2 - CENTER,
   588→   588→    };
   589→   589→    applyTransform();
   590→   590→  }, [applyTransform]);
   591→   591→
   592→   592→  const currentZoom = displayZoom;
   593→   593→
   594→   594→  return (
   595→   595→    <div
   596→   596→      className="w-full h-[65vh] min-h-[500px] overflow-hidden rounded-3xl border border-white/15 shadow-2xl relative bg-black/40 touch-none select-none"
   597→   597→      ref={containerRef}
   598→   598→      onMouseDown={onMouseDown}
   599→   599→      onMouseUp={() => setIsDragging(false)}
   600→   600→      onMouseLeave={() => setIsDragging(false)}
   601→   601→      onMouseMove={onMouseMove}
   602→   602→      style={{ cursor: isDragging ? 'grabbing' : 'grab' }}
   603→   603→    >
   604→   604→      <div
   605→   605→        ref={contentRef}
   606→   606→        className="absolute top-0 left-0"
   607→   607→        style={{
   608→   608→          width: CONTENT_SIZE,
   609→   609→          height: CONTENT_SIZE,
   610→   610→          transformOrigin: '0 0',
   611→   611→          willChange: 'transform',
   612→   612→        }}
   613→   613→      >
   614→   614→        <div className="absolute inset-0 grid-bg opacity-30 pointer-events-none" />
   615→   615→        <div className="absolute inset-0 bg-[radial-gradient(ellipse_at_center,rgba(0,255,255,0.06),transparent_60%)] pointer-events-none" />
   616→   616→        {children}
   617→   617→      </div>
   618→   618→
   619→   619→      {/* Zoom controls overlay */}
   620→   620→      <div className="absolute bottom-3 right-3 flex flex-col gap-1.5 z-40">
   621→   621→        <button
   622→   622→          onClick={() => { audio.playClick(); zoomBy(1.2); }}
   623→   623→          className="w-9 h-9 rounded-lg glass-panel border border-cyan-500/30 text-cyan-300 font-orbitron font-bold text-lg flex items-center justify-center hover:border-cyan-400/60 hover:bg-cyan-500/10 transition-all shadow-[0_0_10px_rgba(0,255,255,0.15)]"
   624→   624→          title={t('upgrades.zoomIn')}
   625→   625→        >
   626→   626→          +
   627→   627→        </button>
   628→   628→        <button
   629→   629→          onClick={() => { audio.playClick(); zoomBy(1 / 1.2); }}
   630→   630→          className="w-9 h-9 rounded-lg glass-panel border border-cyan-500/30 text-cyan-300 font-orbitron font-bold text-lg flex items-center justify-center hover:border-cyan-400/60 hover:bg-cyan-500/10 transition-all shadow-[0_0_10px_rgba(0,255,255,0.15)]"
   631→   631→          title={t('upgrades.zoomOut')}
   632→   632→        >
   633→   633→          −
   634→   634→        </button>
   635→   635→        <button
   636→   636→          onClick={() => { audio.playClick(); resetView(); }}
   637→   637→          className="w-9 h-9 rounded-lg glass-panel border border-cyan-500/30 text-cyan-300 font-orbitron font-bold text-[10px] flex items-center justify-center hover:border-cyan-400/60 hover:bg-cyan-500/10 transition-all shadow-[0_0_10px_rgba(0,255,255,0.15)]"
   638→   638→          title={t('upgrades.zoomReset')}
   639→   639→        >
   640→   640→          1:1
   641→   641→        </button>
   642→   642→      </div>
   643→   643→
   644→   644→      {/* Zoom level indicator */}
   645→   645→      <div className="absolute bottom-3 left-3 z-40 glass-panel px-3 py-1.5 rounded-lg border border-white/10 font-orbitron text-xs text-cyan-300/80 tracking-widest shadow-[0_0_10px_rgba(0,0,0,0.4)]">
   646→   646→        {Math.round(currentZoom * 100)}%
   647→   647→      </div>
   648→   648→    </div>
   649→   649→  );
   650→   650→}
