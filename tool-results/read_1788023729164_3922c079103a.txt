    25→export function renderBackground(ctx: CanvasRenderingContext2D, W: number, H: number, frame: number, cx: number, cy: number) {
    26→  ctx.clearRect(0, 0, W, H);
    27→
    28→  // Deep space gradient
    29→  const bgGrad = ctx.createRadialGradient(cx, cy, 0, cx, cy, Math.max(W, H));
    30→  bgGrad.addColorStop(0, '#0a0520');
    31→  bgGrad.addColorStop(1, '#020108');
    32→  ctx.fillStyle = bgGrad;
    33→  ctx.fillRect(0, 0, W, H);
    34→
    35→  // Nebulae (Gas clouds)
    36→  ctx.globalCompositeOperation = 'screen';
    37→  const t = frame * 0.002;
    38→  const drawNebula = (nx: number, ny: number, r: number, color: string) => {
    39→    const nGrad = ctx.createRadialGradient(nx, ny, 0, nx, ny, r);
    40→    nGrad.addColorStop(0, color);
    41→    nGrad.addColorStop(1, 'transparent');
    42→    ctx.fillStyle = nGrad;
    43→    ctx.beginPath();
    44→    ctx.arc(nx, ny, r, 0, Math.PI * 2);
    45→    ctx.fill();
    46→  };
    47→  drawNebula(W * 0.2 + Math.sin(t) * 150, H * 0.3 + Math.cos(t * 0.8) * 150, Math.max(W, H) * 0.5, 'rgba(100, 0, 255, 0.07)');
    48→  drawNebula(W * 0.8 + Math.cos(t * 1.2) * 150, H * 0.7 + Math.sin(t * 1.1) * 150, Math.max(W, H) * 0.4, 'rgba(0, 200, 255, 0.06)');
    49→  drawNebula(W * 0.5 + Math.sin(t * 0.5) * 200, H * 0.9 + Math.cos(t * 0.7) * 100, Math.max(W, H) * 0.4, 'rgba(255, 0, 100, 0.04)');
    50→  ctx.globalCompositeOperation = 'source-over';
    51→
    52→  // Parallax Stars
    53→  for (let i = 0; i < 150; i++) {
    54→    const origX = (Math.sin(i * 123.45) * 0.5 + 0.5) * W;
    55→    const origY = (Math.cos(i * 678.90) * 0.5 + 0.5) * H;
    56→    
    57→    const layer = i % 3; // 0: far, 1: mid, 2: close
    58→    const speed = (layer + 1) * 0.15;
    59→    const size = (layer === 2 ? 1.5 : 0.8) * (Math.sin(frame * 0.03 + i) * 0.4 + 0.6);
    60→    
    61→    const x = (origX - frame * speed) % W;
    62→    const y = (origY + frame * speed * 0.3) % H;
    63→    
    64→    // Handle negative modulo for smooth wrapping
    65→    const wrapX = x < 0 ? x + W : x;
    66→    const wrapY = y < 0 ? y + H : y;
    67→
    68→    ctx.fillStyle = layer === 2 ? 'rgba(255, 255, 255, 0.9)' : (layer === 1 ? 'rgba(150, 200, 255, 0.6)' : 'rgba(100, 100, 150, 0.3)');
    69→    ctx.beginPath();
    70→    ctx.arc(wrapX, wrapY, size, 0, Math.PI * 2);
    71→    ctx.fill();
    72→    
    73→    // Add a slight glow to close stars
    74→    if (layer === 2 && Math.sin(frame * 0.05 + i) > 0.8) {
    75→      ctx.beginPath();
    76→      ctx.arc(wrapX, wrapY, size * 2.5, 0, Math.PI * 2);
    77→      ctx.fillStyle = 'rgba(255, 255, 255, 0.4)';
    78→      ctx.fill();
    79→    }
    80→  }

    83→  ctx.strokeStyle = 'rgba(0,255,255,0.03)';
    84→  ctx.lineWidth = 1;
    85→  const gridSize = 80;
    86→  const offsetX = (frame * 0.2) % gridSize;
    87→  const offsetY = (frame * 0.2) % gridSize;
    88→
    89→  ctx.beginPath();
    90→  for (let gx = offsetX; gx < W; gx += gridSize) {
    91→    ctx.moveTo(gx, 0); ctx.lineTo(gx, H);
    92→  }
    93→  for (let gy = offsetY; gy < H; gy += gridSize) {
    94→    ctx.moveTo(0, gy); ctx.lineTo(W, gy);
    95→  }
    96→  ctx.stroke();
    97→
    98→  // Center rings with glow
    99→  for (let r = 80; r < Math.max(W, H); r += 150) {
   100→    ctx.beginPath();
   101→    ctx.arc(cx, cy, r, 0, Math.PI * 2);
   102→    const alpha = 0.02 + 0.02 * Math.sin(frame * 0.015 + r * 0.001);
   103→    ctx.strokeStyle = `rgba(0,255,255,${alpha})`;
   104→    ctx.lineWidth = 2;
   105→    ctx.stroke();
   106→  }

   109→export function renderParticles(ctx: CanvasRenderingContext2D, particles: Particle[]) {
   110→  const TAU = Math.PI * 2;
   111→  // Precompute alpha hex strings (0-255 → "00".."ff") to avoid per-particle
   112→  // string concatenation + Math.floor + toString in the hot loop.
   113→  const alphaHex: string[] = new Array(256);
   114→  for (let a = 0; a < 256; a++) {
   115→    alphaHex[a] = a.toString(16).padStart(2, '0');
   116→  }
   117→
   118→  for (let i = 0; i < particles.length; i++) {
   119→    const p = particles[i];
   120→    const alpha = p.life / p.maxLife;
   121→    const r = p.radius * alpha;
   122→
   123→    // Fake glow
   124→    const glowAlphaByte = Math.min(255, Math.floor(alpha * 0.3 * 255));
   125→    ctx.beginPath();
   126→    ctx.arc(p.x, p.y, r * 2.5, 0, TAU);
   127→    ctx.fillStyle = p.color + alphaHex[glowAlphaByte];
   128→    ctx.fill();
   129→
   130→    // Core
   131→    const coreAlphaByte = Math.min(255, Math.floor(alpha * 255));
   132→    ctx.beginPath();
   133→    ctx.arc(p.x, p.y, r, 0, TAU);
   134→    ctx.fillStyle = p.color + alphaHex[coreAlphaByte];
   135→    ctx.fill();
   136→  }
   137→}

   198→export function renderGravityBeams(ctx: CanvasRenderingContext2D, beams: { x1: number; y1: number; x2: number; y2: number; life: number; maxLife: number }[]) {
   199→  // Purple animated energy beams from the boss to enemies.
   200→  // Multiple layers for a volumetric look.
   201→  const TAU = Math.PI * 2;
   202→  const time = Date.now();
   203→
   204→  ctx.save();
   205→  ctx.globalCompositeOperation = 'lighter';
   206→
   207→  for (let i = 0; i < beams.length; i++) {
   208→    const beam = beams[i];
   209→    const alpha = beam.life / beam.maxLife;
   210→    const dx = beam.x2 - beam.x1;
   211→    const dy = beam.y2 - beam.y1;
   212→    const len = Math.sqrt(dx * dx + dy * dy);
   213→    if (len < 1) continue;
   214→    const angle = Math.atan2(dy, dx);
   215→
   216→    ctx.save();
   217→    ctx.translate(beam.x1, beam.y1);
   218→    ctx.rotate(angle);
   219→
   220→    // Layer 1: Wide outer glow
   221→    ctx.beginPath();
   222→    ctx.moveTo(0, 0);
   223→    ctx.lineTo(len, 0);
   224→    ctx.strokeStyle = `rgba(191, 0, 255, ${alpha * 0.15})`;
   225→    ctx.lineWidth = 16;
   226→    ctx.lineCap = 'round';
   227→    ctx.stroke();
   228→
   229→    // Layer 2: Mid beam
   230→    ctx.beginPath();
   231→    ctx.moveTo(0, 0);
   232→    ctx.lineTo(len, 0);
   233→    ctx.strokeStyle = `rgba(191, 0, 255, ${alpha * 0.4})`;
   234→    ctx.lineWidth = 6;
   235→    ctx.stroke();
   236→
   237→    // Layer 3: Animated energy pulses traveling along the beam
   238→    const pulseCount = Math.max(2, Math.floor(len / 40));
   239→    for (let p = 0; p < pulseCount; p++) {
   240→      const phase = ((time / 200) + p / pulseCount) % 1;
   241→      const px = phase * len;
   242→      const pulseR = 4 + Math.sin(time / 100 + p) * 2;
   243→      // glow
   244→      ctx.beginPath();
   245→      ctx.arc(px, 0, pulseR * 2, 0, TAU);
   246→      ctx.fillStyle = `rgba(191, 0, 255, ${alpha * 0.3})`;
   247→      ctx.fill();
   248→      // core
   249→      ctx.beginPath();
   250→      ctx.arc(px, 0, pulseR, 0, TAU);
   251→      ctx.fillStyle = `rgba(255, 255, 255, ${alpha * 0.8})`;
   252→      ctx.fill();
   253→    }
   254→
   255→    // Layer 4: Bright core line
   256→    ctx.beginPath();
   257→    ctx.moveTo(0, 0);
   258→    ctx.lineTo(len, 0);
   259→    ctx.strokeStyle = `rgba(255, 255, 255, ${alpha * 0.6})`;
   260→    ctx.lineWidth = 1.5;
   261→    ctx.stroke();
   262→
   263→    ctx.restore();
   264→  }
   265→
   266→  ctx.restore();
   267→}

   269→export function renderMiniBlackHoles(ctx: CanvasRenderingContext2D, blackHoles: { x: number; y: number; radius: number; absorbRadius: number; life: number; maxLife: number; angle: number }[]) {

   284→
