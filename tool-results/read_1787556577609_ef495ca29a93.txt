   355→              style={{ filter: `drop-shadow(0 0 4px ${category.color})`, transition: 'stroke-dasharray 0.8s ease' }}
   356→            />
   357→          </svg>

   367→          <StatRow key={stat.key} stat={stat} color={category.color} />

   369→      </div>
   370→    </div>
   371→  );

   374→/** Full stats tab: hero summary + categorized stats. */
   375→function StatsTab({ progress }: { progress: GameProgress }) {
   376→  const categories = buildStatCategories(progress.upgrades);
   377→  const allStats = categories.flatMap(c => c.stats);
   378→  const totalLevels = allStats.reduce((s, st) => s + st.level, 0);
   379→  const maxLevels = allStats.reduce((s, st) => s + st.maxLevel, 0);
   380→  const overallPct = maxLevels > 0 ? (totalLevels / maxLevels) * 100 : 0;
   381→
   382→  // Combat Power — weighted aggregate for a satisfying single number
   383→  const u = (k: keyof GameProgress['upgrades']) => progress.upgrades[k] || 0;
   384→  const combatPower = Math.round(
   385→    (100 + u('maxHp') * 5) * (1 + u('damage') * 0.15) * (1 + u('critChance') * 0.03) * (1 + u('critDamage') * 0.075)
   386→    * (1 + u('maxArrows') * 0.08) * (1 + u('projectileSpeed') * 0.05) * (1 + u('piercing') * 0.15)
   387→    * (1 + u('defense') * 0.03) * (1 + u('reloadSpeed') * 0.05) * (1 + u('regenSpeed') * 0.04)
   388→  );
   389→
   390→  const baseUnlocked = u('baseNode') > 0;
   391→
   392→  return (
   393→    <div className="w-full max-w-4xl mx-auto mt-4 mb-16 animate-fade-in">
   394→      {/* Hero summary panel */}

   463→/**
   464→ * Pan + wheel-zoom canvas for the upgrade tree.
   465→ * Defined at module scope (NOT inside UpgradesScreen) so it keeps a stable
   466→ * component identity across re-renders — otherwise its refs/transform would
   467→ * reset every time the parent updates state.
   468→ */
   469→function UpgradeTreeCanvas({
   470→  children,
   471→  onZoomChange,
   472→}: {
   473→  children: React.ReactNode;
   474→  onZoomChange: (zoom: number) => void;
   475→}) {
   476→  const containerRef = useRef<HTMLDivElement>(null);
   477→  const contentRef = useRef<HTMLDivElement>(null);
   478→  const transformRef = useRef({ zoom: 1, panX: 0, panY: 0 });
   479→  // displayZoom mirrors the ref into state so we can read it during render
   480→  // for the zoom indicator (refs must not be read during render).
   481→  const [displayZoom, setDisplayZoom] = useState(1);
   482→  const [isDragging, setIsDragging] = useState(false);
   483→  const dragStart = useRef({ x: 0, y: 0, panX: 0, panY: 0, moved: false });
   484→
   485→  const applyTransform = useCallback(() => {
   486→    const el = contentRef.current;
   487→    if (el) {
   488→      const { zoom, panX, panY } = transformRef.current;
   489→      el.style.transform = `translate(${panX}px, ${panY}px) scale(${zoom})`;
   490→    }
   491→    onZoomChange(transformRef.current.zoom);
   492→    setDisplayZoom(transformRef.current.zoom);
   493→  }, [onZoomChange]);
   494→
   495→  // Center the content on first mount
   496→  useEffect(() => {
   497→    const el = containerRef.current;
   498→    if (!el) return;
   499→    const cw = el.clientWidth;
   500→    const ch = el.clientHeight;
   501→    transformRef.current = {
   502→      zoom: 1,
   503→      panX: cw / 2 - CENTER,
   504→      panY: ch / 2 - CENTER,
   505→    };
   506→    applyTransform();
   507→    const handleResize = () => {
   508→      const z = transformRef.current.zoom;
   509→      transformRef.current.panX = el.clientWidth / 2 - CENTER * z;
   510→      transformRef.current.panY = el.clientHeight / 2 - CENTER * z;
   511→      applyTransform();
   512→    };
   513→    window.addEventListener('resize', handleResize);
   514→    return () => window.removeEventListener('resize', handleResize);
   515→  }, [applyTransform]);
   516→
   517→  // Non-passive wheel listener so we can preventDefault page scroll
   518→  useEffect(() => {
   519→    const el = containerRef.current;
   520→    if (!el) return;
   521→    const onWheel = (e: WheelEvent) => {
   522→      e.preventDefault();
   523→      const rect = el.getBoundingClientRect();
   524→      const cx = e.clientX - rect.left;
   525→      const cy = e.clientY - rect.top;
   526→      const { zoom: prevZoom, panX: prevPanX, panY: prevPanY } = transformRef.current;
   527→      const factor = e.deltaY < 0 ? 1.12 : 1 / 1.12;
   528→      const newZoom = clampZoom(prevZoom * factor);
   529→      const actualFactor = newZoom / prevZoom;
   530→      transformRef.current.zoom = newZoom;
   531→      transformRef.current.panX = cx - (cx - prevPanX) * actualFactor;
   532→      transformRef.current.panY = cy - (cy - prevPanY) * actualFactor;
   533→      applyTransform();
   534→    };
   535→    el.addEventListener('wheel', onWheel, { passive: false });
   536→    return () => el.removeEventListener('wheel', onWheel);
   537→  }, [applyTransform]);
   538→
   539→  const onMouseDown = (e: React.MouseEvent) => {
   540→    if (e.button !== 0) return;
   541→    dragStart.current = {
   542→      x: e.clientX,
   543→      y: e.clientY,
   544→      panX: transformRef.current.panX,
   545→      panY: transformRef.current.panY,
   546→      moved: false,
   547→    };
   548→    setIsDragging(true);
   549→  };
   550→
   551→  const onMouseMove = (e: React.MouseEvent) => {
   552→    if (!isDragging) return;
   553→    const dx = e.clientX - dragStart.current.x;
   554→    const dy = e.clientY - dragStart.current.y;
   555→    if (Math.abs(dx) > 3 || Math.abs(dy) > 3) dragStart.current.moved = true;
   556→    transformRef.current.panX = dragStart.current.panX + dx;
   557→    transformRef.current.panY = dragStart.current.panY + dy;
   558→    applyTransform();
   559→  };
   560→
   561→  const zoomBy = useCallback((factor: number) => {
   562→    const el = containerRef.current;
   563→    if (!el) return;
   564→    const cx = el.clientWidth / 2;
   565→    const cy = el.clientHeight / 2;
   566→    const { zoom: prevZoom, panX: prevPanX, panY: prevPanY } = transformRef.current;
   567→    const newZoom = clampZoom(prevZoom * factor);
   568→    const actualFactor = newZoom / prevZoom;
   569→    transformRef.current.zoom = newZoom;
   570→    transformRef.current.panX = cx - (cx - prevPanX) * actualFactor;
   571→    transformRef.current.panY = cy - (cy - prevPanY) * actualFactor;
   572→    applyTransform();
   573→  }, [applyTransform]);
   574→
   575→  const resetView = useCallback(() => {
   576→    const el = containerRef.current;
   577→    if (!el) return;
   578→    transformRef.current = {
   579→      zoom: 1,
   580→      panX: el.clientWidth / 2 - CENTER,
   581→      panY: el.clientHeight / 2 - CENTER,
   582→    };
   583→    applyTransform();
   584→  }, [applyTransform]);
   585→
   586→  const currentZoom = displayZoom;
   587→
   588→  return (
   589→    <div
   590→      className="w-full h-[65vh] min-h-[500px] overflow-hidden rounded-3xl border border-white/15 shadow-2xl relative bg-black/40 touch-none select-none"
   591→      ref={containerRef}
   592→      onMouseDown={onMouseDown}
   593→      onMouseUp={() => setIsDragging(false)}
   594→      onMouseLeave={() => setIsDragging(false)}
   595→      onMouseMove={onMouseMove}
   596→      style={{ cursor: isDragging ? 'grabbing' : 'grab' }}
   597→    >
   598→      <div
   599→        ref={contentRef}
   600→        className="absolute top-0 left-0"
   601→        style={{
   602→          width: CONTENT_SIZE,
   603→          height: CONTENT_SIZE,
   604→          transformOrigin: '0 0',
   605→          willChange: 'transform',
   606→        }}
   607→      >
   608→        <div className="absolute inset-0 grid-bg opacity-30 pointer-events-none" />
   609→        <div className="absolute inset-0 bg-[radial-gradient(ellipse_at_center,rgba(0,255,255,0.06),transparent_60%)] pointer-events-none" />
   610→        {children}
   611→      </div>
   612→
   613→      {/* Zoom controls overlay */}
   614→      <div className="absolute bottom-3 right-3 flex flex-col gap-1.5 z-40">
   615→        <button
   616→          onClick={() => { audio.playClick(); zoomBy(1.2); }}
   617→          className="w-9 h-9 rounded-lg glass-panel border border-cyan-500/30 text-cyan-300 font-orbitron font-bold text-lg flex items-center justify-center hover:border-cyan-400/60 hover:bg-cyan-500/10 transition-all shadow-[0_0_10px_rgba(0,255,255,0.15)]"
   618→          title="Приблизить"
   619→        >
   620→          +
   621→        </button>
   622→        <button
   623→          onClick={() => { audio.playClick(); zoomBy(1 / 1.2); }}
   624→          className="w-9 h-9 rounded-lg glass-panel border border-cyan-500/30 text-cyan-300 font-orbitron font-bold text-lg flex items-center justify-center hover:border-cyan-400/60 hover:bg-cyan-500/10 transition-all shadow-[0_0_10px_rgba(0,255,255,0.15)]"
   625→          title="Отдалить"
   626→        >
   627→          −
   628→        </button>
   629→        <button
   630→          onClick={() => { audio.playClick(); resetView(); }}
   631→          className="w-9 h-9 rounded-lg glass-panel border border-cyan-500/30 text-cyan-300 font-orbitron font-bold text-[10px] flex items-center justify-center hover:border-cyan-400/60 hover:bg-cyan-500/10 transition-all shadow-[0_0_10px_rgba(0,255,255,0.15)]"
   632→          title="Сбросить вид"
   633→        >
   634→          1:1
   635→        </button>
   636→      </div>
   637→
   638→      {/* Zoom level indicator */}
   639→      <div className="absolute bottom-3 left-3 z-40 glass-panel px-3 py-1.5 rounded-lg border border-white/10 font-orbitron text-xs text-cyan-300/80 tracking-widest shadow-[0_0_10px_rgba(0,0,0,0.4)]">
   640→        {Math.round(currentZoom * 100)}%
   641→      </div>
   642→    </div>
   643→  );
   644→}
