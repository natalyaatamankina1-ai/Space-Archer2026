   763→/**
   764→ * Pan + wheel-zoom canvas for the upgrade tree.
   765→ * Defined at module scope (NOT inside UpgradesScreen) so it keeps a stable
   766→ * component identity across re-renders — otherwise its refs/transform would
   767→ * reset every time the parent updates state.
   768→ */
   769→function UpgradeTreeCanvas({
   770→  children,
   771→  onZoomChange,
   772→}: {
   773→  children: React.ReactNode;
   774→  onZoomChange: (zoom: number) => void;
   775→}) {
   776→  const { t } = useLanguage();
   777→  const containerRef = useRef<HTMLDivElement>(null);
   778→  const contentRef = useRef<HTMLDivElement>(null);
   779→  const transformRef = useRef({ zoom: 1, panX: 0, panY: 0 });
   780→  // displayZoom mirrors the ref into state so we can read it during render
   781→  // for the zoom indicator (refs must not be read during render).
   782→  const [displayZoom, setDisplayZoom] = useState(1);
   783→  const [isDragging, setIsDragging] = useState(false);
   784→  const dragStart = useRef({ x: 0, y: 0, panX: 0, panY: 0, moved: false });
   785→
   786→  const applyTransform = useCallback(() => {
   787→    const el = contentRef.current;
   788→    if (el) {
   789→      const { zoom, panX, panY } = transformRef.current;
   790→      el.style.transform = `translate(${panX}px, ${panY}px) scale(${zoom})`;
   791→    }
   792→    onZoomChange(transformRef.current.zoom);
   793→    setDisplayZoom(transformRef.current.zoom);
   794→  }, [onZoomChange]);
   795→
   796→  // Center the content on first mount
   797→  useEffect(() => {
   798→    const el = containerRef.current;
   799→    if (!el) return;
   800→    const cw = el.clientWidth;
   801→    const ch = el.clientHeight;
   802→    transformRef.current = {
   803→      zoom: 1,
   804→      panX: cw / 2 - CENTER,
   805→      panY: ch / 2 - CENTER,
   806→    };
   807→    applyTransform();
   808→    const handleResize = () => {
   809→      const z = transformRef.current.zoom;
   810→      transformRef.current.panX = el.clientWidth / 2 - CENTER * z;
   811→      transformRef.current.panY = el.clientHeight / 2 - CENTER * z;
   812→      applyTransform();
   813→    };
   814→    window.addEventListener('resize', handleResize);
   815→    return () => window.removeEventListener('resize', handleResize);
   816→  }, [applyTransform]);
   817→
   818→  // Non-passive wheel listener so we can preventDefault page scroll
   819→  useEffect(() => {
   820→    const el = containerRef.current;
   821→    if (!el) return;
   822→    const onWheel = (e: WheelEvent) => {
   823→      e.preventDefault();
   824→      const rect = el.getBoundingClientRect();
   825→      const cx = e.clientX - rect.left;
   826→      const cy = e.clientY - rect.top;
   827→      const { zoom: prevZoom, panX: prevPanX, panY: prevPanY } = transformRef.current;
   828→      const factor = e.deltaY < 0 ? 1.12 : 1 / 1.12;
   829→      const newZoom = clampZoom(prevZoom * factor);
   830→      const actualFactor = newZoom / prevZoom;
   831→      transformRef.current.zoom = newZoom;
   832→      transformRef.current.panX = cx - (cx - prevPanX) * actualFactor;
   833→      transformRef.current.panY = cy - (cy - prevPanY) * actualFactor;
   834→      applyTransform();
   835→    };
   836→    el.addEventListener('wheel', onWheel, { passive: false });
   837→    return () => el.removeEventListener('wheel', onWheel);
   838→  }, [applyTransform]);
   839→
   840→  const onMouseDown = (e: React.MouseEvent) => {
   841→    if (e.button !== 0) return;
   842→    dragStart.current = {
   843→      x: e.clientX,
   844→      y: e.clientY,
   845→      panX: transformRef.current.panX,
   846→      panY: transformRef.current.panY,
   847→      moved: false,
   848→    };
   849→    setIsDragging(true);
   850→  };
   851→
   852→  const onMouseMove = (e: React.MouseEvent) => {
   853→    if (!isDragging) return;
