   320→   320→function StatCategoryBlock({ category }: { category: StatCategoryDef }) {
   321→   321→  const acquired = category.stats.filter(s => s.level > 0).length;
   322→   322→  const totalLevels = category.stats.reduce((sum, s) => sum + s.level, 0);
   323→   323→  const maxLevels = category.stats.reduce((sum, s) => sum + s.maxLevel, 0);
   324→   324→  const pct = maxLevels > 0 ? (totalLevels / maxLevels) * 100 : 0;
   325→   325→
   326→   326→  return (
   327→   327→    <div className="mb-8">
   328→   328→      {/* Category header */}
   329→   329→      <div className="flex items-center gap-3 mb-4">
   330→   330→        <div
   331→   331→          className="w-10 h-10 rounded-lg flex items-center justify-center text-xl border shrink-0"
   332→   332→          style={{
   333→   333→            background: `${category.color}15`,
   334→   334→            borderColor: `${category.color}44`,
   335→   335→            textShadow: `0 0 12px ${category.color}`,
   336→   336→          }}
   337→   337→        >
   338→   338→          {category.icon}
   339→   339→        </div>
   340→   340→        <div className="flex-1">
   341→   341→          <div className="font-orbitron font-black text-lg tracking-wider" style={{ color: category.color, textShadow: `0 0 8px ${category.color}66` }}>
   342→   342→            {category.name}
   343→   343→          </div>
   344→   344→          <div className="font-rajdhani text-[11px] text-white/40 tracking-widest">

   463→   463→/**
   464→   464→ * Pan + wheel-zoom canvas for the upgrade tree.
   465→   465→ * Defined at module scope (NOT inside UpgradesScreen) so it keeps a stable
   466→   466→ * component identity across re-renders — otherwise its refs/transform would
   467→   467→ * reset every time the parent updates state.
   468→   468→ */
   469→   469→function UpgradeTreeCanvas({
   470→   470→  children,
   471→   471→  onZoomChange,
   472→   472→}: {
   473→   473→  children: React.ReactNode;
   474→   474→  onZoomChange: (zoom: number) => void;
   475→   475→}) {
   476→   476→  const containerRef = useRef<HTMLDivElement>(null);
   477→   477→  const contentRef = useRef<HTMLDivElement>(null);
   478→   478→  const transformRef = useRef({ zoom: 1, panX: 0, panY: 0 });
   479→   479→  // displayZoom mirrors the ref into state so we can read it during render
   480→   480→  // for the zoom indicator (refs must not be read during render).
   481→   481→  const [displayZoom, setDisplayZoom] = useState(1);
   482→   482→  const [isDragging, setIsDragging] = useState(false);
   483→   483→  const dragStart = useRef({ x: 0, y: 0, panX: 0, panY: 0, moved: false });
   484→   484→
   485→   485→  const applyTransform = useCallback(() => {
   486→   486→    const el = contentRef.current;
   487→   487→    if (el) {
   488→   488→      const { zoom, panX, panY } = transformRef.current;
   489→   489→      el.style.transform = `translate(${panX}px, ${panY}px) scale(${zoom})`;
   490→   490→    }
   491→   491→    onZoomChange(transformRef.current.zoom);
   492→   492→    setDisplayZoom(transformRef.current.zoom);
   493→   493→  }, [onZoomChange]);
   494→   494→
   495→   495→  // Center the content on first mount
   496→   496→  useEffect(() => {
   497→   497→    const el = containerRef.current;
   498→   498→    if (!el) return;
   499→   499→    const cw = el.clientWidth;
   500→   500→    const ch = el.clientHeight;
   501→   501→    transformRef.current = {
   502→   502→      zoom: 1,
   503→   503→      panX: cw / 2 - CENTER,
   504→   504→      panY: ch / 2 - CENTER,
   505→   505→    };
   506→   506→    applyTransform();
   507→   507→    const handleResize = () => {
   508→   508→      const z = transformRef.current.zoom;
   509→   509→      transformRef.current.panX = el.clientWidth / 2 - CENTER * z;
   510→   510→      transformRef.current.panY = el.clientHeight / 2 - CENTER * z;
   511→   511→      applyTransform();
   512→   512→    };
   513→   513→    window.addEventListener('resize', handleResize);
   514→   514→    return () => window.removeEventListener('resize', handleResize);
   515→   515→  }, [applyTransform]);
   516→   516→
   517→   517→  // Non-passive wheel listener so we can preventDefault page scroll
   518→   518→  useEffect(() => {
   519→   519→    const el = containerRef.current;
   520→   520→    if (!el) return;
   521→   521→    const onWheel = (e: WheelEvent) => {
   522→   522→      e.preventDefault();
   523→   523→      const rect = el.getBoundingClientRect();
   524→   524→      const cx = e.clientX - rect.left;
   525→   525→      const cy = e.clientY - rect.top;
   526→   526→      const { zoom: prevZoom, panX: prevPanX, panY: prevPanY } = transformRef.current;
   527→   527→      const factor = e.deltaY < 0 ? 1.12 : 1 / 1.12;
   528→   528→      const newZoom = clampZoom(prevZoom * factor);
   529→   529→      const actualFactor = newZoom / prevZoom;
   530→   530→      transformRef.current.zoom = newZoom;
   531→   531→      transformRef.current.panX = cx - (cx - prevPanX) * actualFactor;
   532→   532→      transformRef.current.panY = cy - (cy - prevPanY) * actualFactor;
   533→   533→      applyTransform();
   534→   534→    };
   535→   535→    el.addEventListener('wheel', onWheel, { passive: false });
   536→   536→    return () => el.removeEventListener('wheel', onWheel);
   537→   537→  }, [applyTransform]);
   538→   538→
   539→   539→  const onMouseDown = (e: React.MouseEvent) => {
   540→   540→    if (e.button !== 0) return;
   541→   541→    dragStart.current = {
   542→   542→      x: e.clientX,
   543→   543→      y: e.clientY,
   544→   544→      panX: transformRef.current.panX,
   545→   545→      panY: transformRef.current.panY,
   546→   546→      moved: false,
   547→   547→    };
   548→   548→    setIsDragging(true);
   549→   549→  };
   550→   550→
   551→   551→  const onMouseMove = (e: React.MouseEvent) => {
   552→   552→    if (!isDragging) return;
   553→   553→    const dx = e.clientX - dragStart.current.x;
   554→   554→    const dy = e.clientY - dragStart.current.y;
   555→   555→    if (Math.abs(dx) > 3 || Math.abs(dy) > 3) dragStart.current.moved = true;
   556→   556→    transformRef.current.panX = dragStart.current.panX + dx;
   557→   557→    transformRef.current.panY = dragStart.current.panY + dy;
   558→   558→    applyTransform();
   559→   559→  };
   560→   560→
   561→   561→  const zoomBy = useCallback((factor: number) => {
   562→   562→    const el = containerRef.current;
   563→   563→    if (!el) return;
   564→   564→    const cx = el.clientWidth / 2;
   565→   565→    const cy = el.clientHeight / 2;
   566→   566→    const { zoom: prevZoom, panX: prevPanX, panY: prevPanY } = transformRef.current;
   567→   567→    const newZoom = clampZoom(prevZoom * factor);
   568→   568→    const actualFactor = newZoom / prevZoom;
   569→   569→    transformRef.current.zoom = newZoom;
   570→   570→    transformRef.current.panX = cx - (cx - prevPanX) * actualFactor;
   571→   571→    transformRef.current.panY = cy - (cy - prevPanY) * actualFactor;
   572→   572→    applyTransform();
   573→   573→  }, [applyTransform]);
   574→   574→
   575→   575→  const resetView = useCallback(() => {
   576→   576→    const el = containerRef.current;
   577→   577→    if (!el) return;
   578→   578→    transformRef.current = {
   579→   579→      zoom: 1,
   580→   580→      panX: el.clientWidth / 2 - CENTER,
   581→   581→      panY: el.clientHeight / 2 - CENTER,
   582→   582→    };
   583→   583→    applyTransform();
   584→   584→  }, [applyTransform]);
   585→   585→
   586→   586→  const currentZoom = displayZoom;
   587→   587→
   588→   588→  return (
   589→   589→    <div
   590→   590→      className="w-full h-[65vh] min-h-[500px] overflow-hidden rounded-3xl border border-white/15 shadow-2xl relative bg-black/40 touch-none select-none"
   591→   591→      ref={containerRef}
   592→   592→      onMouseDown={onMouseDown}
   593→   593→      onMouseUp={() => setIsDragging(false)}
   594→   594→      onMouseLeave={() => setIsDragging(false)}
   595→   595→      onMouseMove={onMouseMove}
   596→   596→      style={{ cursor: isDragging ? 'grabbing' : 'grab' }}
   597→   597→    >
   598→   598→      <div
   599→   599→        ref={contentRef}
   600→   600→        className="absolute top-0 left-0"
   601→   601→        style={{
   602→   602→          width: CONTENT_SIZE,
   603→   603→          height: CONTENT_SIZE,
   604→   604→          transformOrigin: '0 0',
   605→   605→          willChange: 'transform',
   606→   606→        }}
   607→   607→      >
   608→   608→        <div className="absolute inset-0 grid-bg opacity-30 pointer-events-none" />
   609→   609→        <div className="absolute inset-0 bg-[radial-gradient(ellipse_at_center,rgba(0,255,255,0.06),transparent_60%)] pointer-events-none" />
   610→   610→        {children}
   611→   611→      </div>
   612→   612→
   613→   613→      {/* Zoom controls overlay */}
   614→   614→      <div className="absolute bottom-3 right-3 flex flex-col gap-1.5 z-40">
   615→   615→        <button
   616→   616→          onClick={() => { audio.playClick(); zoomBy(1.2); }}
   617→   617→          className="w-9 h-9 rounded-lg glass-panel border border-cyan-500/30 text-cyan-300 font-orbitron font-bold text-lg flex items-center justify-center hover:border-cyan-400/60 hover:bg-cyan-500/10 transition-all shadow-[0_0_10px_rgba(0,255,255,0.15)]"
   618→   618→          title="Приблизить"
   619→   619→        >
   620→   620→          +
   621→   621→        </button>
   622→   622→        <button
   623→   623→          onClick={() => { audio.playClick(); zoomBy(1 / 1.2); }}
   624→   624→          className="w-9 h-9 rounded-lg glass-panel border border-cyan-500/30 text-cyan-300 font-orbitron font-bold text-lg flex items-center justify-center hover:border-cyan-400/60 hover:bg-cyan-500/10 transition-all shadow-[0_0_10px_rgba(0,255,255,0.15)]"
   625→   625→          title="Отдалить"
   626→   626→        >
   627→   627→          −
   628→   628→        </button>
   629→   629→        <button
   630→   630→          onClick={() => { audio.playClick(); resetView(); }}
   631→   631→          className="w-9 h-9 rounded-lg glass-panel border border-cyan-500/30 text-cyan-300 font-orbitron font-bold text-[10px] flex items-center justify-center hover:border-cyan-400/60 hover:bg-cyan-500/10 transition-all shadow-[0_0_10px_rgba(0,255,255,0.15)]"
   632→   632→          title="Сбросить вид"
   633→   633→        >
   634→   634→          1:1
   635→   635→        </button>
   636→   636→      </div>
   637→   637→
   638→   638→      {/* Zoom level indicator */}
   639→   639→      <div className="absolute bottom-3 left-3 z-40 glass-panel px-3 py-1.5 rounded-lg border border-white/10 font-orbitron text-xs text-cyan-300/80 tracking-widest shadow-[0_0_10px_rgba(0,0,0,0.4)]">
   640→   640→        {Math.round(currentZoom * 100)}%
   641→   641→      </div>
   642→   642→    </div>
   643→   643→  );
