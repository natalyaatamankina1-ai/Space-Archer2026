    33→/** Обводка полигональной формы врага (общая для рендера и запекания). */
    34→function tracePath(ctx: CanvasRenderingContext2D, shape: string, r: number) {
    35→  switch (shape) {
    36→    case 'circle':
    37→      ctx.arc(0, 0, r, 0, Math.PI * 2);
    38→      break;
    39→    case 'triangle':
    40→      for (let i = 0; i < 3; i++) {
    41→        const a = (i * Math.PI * 2) / 3 - Math.PI / 2;
    42→        if (i === 0) ctx.moveTo(Math.cos(a) * r, Math.sin(a) * r);
    43→        else ctx.lineTo(Math.cos(a) * r, Math.sin(a) * r);
    44→      }
    45→      ctx.closePath();
    46→      break;
    47→    case 'square':
    48→      ctx.rect(-r * 0.8, -r * 0.8, r * 1.6, r * 1.6);
    49→      break;
    50→    case 'diamond':
    51→      ctx.moveTo(0, -r);
    52→      ctx.lineTo(r * 0.8, 0);
    53→      ctx.lineTo(0, r);
    54→      ctx.lineTo(-r * 0.8, 0);
    55→      ctx.closePath();
    56→      break;
    57→    case 'hexagon':
    58→      for (let i = 0; i < 6; i++) {
    59→        const a = (i * Math.PI * 2) / 6;
    60→        if (i === 0) ctx.moveTo(Math.cos(a) * r, Math.sin(a) * r);
    61→        else ctx.lineTo(Math.cos(a) * r, Math.sin(a) * r);
    62→      }
    63→      ctx.closePath();
    64→      break;
    65→    case 'pentagon':
    66→      for (let i = 0; i < 5; i++) {
    67→        const a = (i * Math.PI * 2) / 5 - Math.PI / 2;
    68→        if (i === 0) ctx.moveTo(Math.cos(a) * r, Math.sin(a) * r);
    69→        else ctx.lineTo(Math.cos(a) * r, Math.sin(a) * r);
    70→      }
    71→      ctx.closePath();
    72→      break;
    73→    case 'star':
    74→      for (let i = 0; i < 10; i++) {
    75→        const a = (i * Math.PI) / 5 - Math.PI / 2;
    76→        const rad = i % 2 === 0 ? r : r * 0.5;
    77→        if (i === 0) ctx.moveTo(Math.cos(a) * rad, Math.sin(a) * rad);
    78→        else ctx.lineTo(Math.cos(a) * rad, Math.sin(a) * rad);
    79→      }
    80→      ctx.closePath();
    81→      break;
    82→    default:
    83→      ctx.arc(0, 0, r, 0, Math.PI * 2);
    84→  }
    85→}

    87→/** Запечённое ядро врага: градиентная заливка + glow-обводка + белая обводка. */
    88→function getShapeSprite(shape: string, color: string, coreRadius: number): HTMLCanvasElement {
    89→  const key = shape + '|' + color + '|' + coreRadius.toFixed(1);
    90→  const cached = shapeSpriteCache.get(key);
    91→  if (cached) return cached;
    92→
    93→  const margin = 12; // запас под широкую glow-обводку (lineWidth 15)
    94→  const size = Math.ceil(coreRadius * 2 + margin * 2);
    95→  const cv = document.createElement('canvas');
    96→  cv.width = size;
    97→  cv.height = size;
    98→  const c = cv.getContext('2d')!;
    99→  c.translate(size / 2, size / 2);
   100→
   101→  // Форма + радиальный градиент (как в оригинале: белый → цвет)
   102→  c.beginPath();
   103→  tracePath(c, shape, coreRadius);
   104→  const grad = c.createRadialGradient(0, 0, 0, 0, 0, coreRadius);
   105→  grad.addColorStop(0, '#ffffff');
   106→  grad.addColorStop(0.3, color);
   107→  grad.addColorStop(1, color);
   108→  c.fillStyle = grad;
   109→  c.fill();
   110→
   111→  // Широкая мягкая обводка-глоу + тонкая белая (как в оригинале)
   112→  c.strokeStyle = color + '44';
   113→  c.lineWidth = 15;
   114→  c.stroke();
   115→  c.strokeStyle = '#ffffff';
   116→  c.lineWidth = 1.5;
   117→  c.stroke();
   118→
   119→  shapeSpriteCache.set(key, cv);
   120→  return cv;
   121→}

   390→export function createEnemy(
   391→  type: Omit<EnemyType, 'hp'>,
   392→  canvasW: number,
   393→  canvasH: number,
   394→  overrides?: Partial<Enemy>
   395→): Enemy {
   396→  if (!type) {
   397→    type = ENEMY_TYPES.find(t => t.id === 'scout')!;
   398→  }
   399→  const side = Math.floor(Math.random() * 4);
   400→  let spawnX = 0, spawnY = 0;
   401→  const margin = 60;
   402→  switch (side) {
   403→    case 0: spawnX = Math.random() * canvasW; spawnY = -margin; break;
   404→    case 1: spawnX = canvasW + margin; spawnY = Math.random() * canvasH; break;
   405→    case 2: spawnX = Math.random() * canvasW; spawnY = canvasH + margin; break;
   406→    case 3: spawnX = -margin; spawnY = Math.random() * canvasH; break;
   407→  }
   408→
   409→  const finalX = overrides?.x ?? spawnX;
   410→  const finalY = overrides?.y ?? spawnY;
   411→  const speed = type.speed * (0.85 + Math.random() * 0.3);
   412→  
   413→  let angle;
   414→  if (type.id === 'bouncer' && !overrides) {
   415→    // Ricochet should aim at a wall, not the player
   416→    const targetSide = (side + 1 + Math.floor(Math.random() * 3)) % 4;
   417→    let tx = 0, ty = 0;
   418→    switch (targetSide) {
   419→      case 0: tx = Math.random() * canvasW; ty = 0; break;
   420→      case 1: tx = canvasW; ty = Math.random() * canvasH; break;
   421→      case 2: tx = Math.random() * canvasW; ty = canvasH; break;
   422→      case 3: tx = 0; ty = Math.random() * canvasH; break;
   423→    }
   424→    angle = Math.atan2(ty - finalY, tx - finalX);
   425→  } else {
   426→    // Default: aim at player (center)
   427→    angle = Math.atan2(canvasH / 2 - finalY, canvasW / 2 - finalX);
   428→  }
   429→
   430→  return {
   431→    ...type,
   432→    hp: type.maxHp,
   433→    x: finalX,
   434→    y: finalY,
   435→    vx: Math.cos(angle) * speed,
   436→    vy: Math.sin(angle) * speed,
   437→    angle,
   438→    trail: [],
   439→    zigzagTimer: 0,
   440→    teleportTimer: type.id === 'phantom' ? Math.random() * 60 : Math.random() * 180,
   441→    splitDone: false,
   442→    pulseRadius: type.radius,
   443→    pulseGrowing: true,
   444→    flashTimer: 0,
   445→    bouncerEntered: false,
   446→    teleportAlpha: (type.pattern === 'teleport' || type.pattern === 'stealth') ? 0 : 1,
   447→    teleportFading: false,
   448→    teleportsLeft: type.pattern === 'teleport' ? 2 + Math.floor(Math.random() * 3) : 0,
   449→    teleportRingAnim: 0,
   450→    ...overrides,
   451→  };
   452→}

   454→export function drawShape(
   455→  ctx: CanvasRenderingContext2D,
   456→  shape: EnemyType['shape'],
   457→  x: number,
   458→  y: number,
   459→  radius: number,
   460→  color: string,
   461→  glowColor: string,
   462→  rotation: number = 0,
   463→  flashTimer: number = 0,
   464→  hp: number = 0,
   465→  maxHp: number = 1,
   466→) {
   467→  ctx.save();
   468→  ctx.translate(x, y);
   469→  ctx.rotate(rotation);
   470→
   471→  const flashAlpha = flashTimer > 0 ? 1 : 0;
   472→  
   473→  if (shape === 'blackhole') {
   474→    // ===== Epic Boss "Omega" render — layered cosmic horror (heavily optimized) =====
   475→    const time = Date.now();
   476→    // Phase indicator: 0 (full hp) → 2 (near death), intensifies visuals.
   477→    // Guard against invalid/missing hp values (defaults: hp=0, maxHp=1 → treat as full).
   478→    const hpRatio = (maxHp > 0 && hp > 0) ? hp / maxHp : 1;
   479→    const phase = Math.min(2, Math.floor((1 - hpRatio) * 3));
   480→    const danger = 0.3 + phase * 0.35; // 0.3 → 1.0 as boss takes damage
   481→
   482→    // Pulsing factor
   483→    const pulse = 1 + Math.sin(time / 280) * 0.04;
   484→
   485→    // Precomputed phase-dependent colors (single source of truth)
   486→    const isCritical = phase >= 2;
   487→    const isDamaged = phase >= 1;
   488→    const haloR = isCritical ? 255 : 128;
   489→    const haloG = isCritical ? 30 : 0;
   490→    const haloB = isCritical ? 60 : 255;
   491→    const accentColor = isCritical ? '#ff0044' : (isDamaged ? '#ff0066' : '#ffffff');
   492→    const sparkColor = isDamaged ? '#ffccff' : '#ffffff';
   493→    const rimColor = isCritical ? 'rgba(255, 51, 102, 0.9)' : 'rgba(255, 255, 255, 0.85)';
   494→    const rimGlowColor = isCritical ? '#ff0044' : glowColor;
   495→    const diskInnerColor = isDamaged ? '#ff66aa' : '#dd99ff';
   496→    const photonRingColor = isCritical ? '#ff3366' : '#ffffff';
   497→
   498→    // ---- Precomputed geometry (avoid recomputing in loops) ----
   499→    const lensR = radius * 3.2 * pulse;
   500→    const diskR = radius * 2.6;
   501→    const midRingR = radius * 2.25;
   502→    const photonR = radius * 1.95;
   503→    const perpR = radius * 2.9;
   504→    const perpInnerR = radius * 2.5;
   505→    const eventR = radius * 1.15 * pulse;
   506→    const rimR = radius * 1.05 * pulse;
   507→    const coreR = radius * 0.55 * pulse;
   508→    const TAU = Math.PI * 2;
   509→
   510→    // Layer 0 — Massive outer gravitational lens halo (color shifts red on damage)
   511→    // Gradient is unavoidable here (needs radial fade), but only ONE fill.
   512→    ctx.save();
   513→    ctx.globalCompositeOperation = 'lighter';
   514→    const lensGrad = ctx.createRadialGradient(0, 0, radius * 1.2, 0, 0, lensR);
   515→    lensGrad.addColorStop(0, `rgba(${haloR}, ${haloG}, ${haloB}, ${0.18 + danger * 0.1})`);
   516→    lensGrad.addColorStop(0.5, `rgba(${haloR}, ${haloG}, ${haloB}, ${0.08 + danger * 0.05})`);
   517→    lensGrad.addColorStop(1, 'rgba(0,0,0,0)');
   518→    ctx.beginPath();
   519→    ctx.arc(0, 0, lensR, 0, TAU);
   520→    ctx.fillStyle = lensGrad;
   521→    ctx.fill();
   522→    ctx.restore();
   523→

   598→    for (let i = 0; i < sparkCount; i++) {
   599→      const sa = rotation * 2.5 + (TAU / sparkCount) * i;
   600→      const sr = radius * (1.8 + Math.sin(time / 200 + i) * 0.4);
   601→      const sparkR = 3 + Math.sin(time / 120 + i * 1.7) * 1.5;

   609→    ctx.beginPath();

   612→    voidGrad.addColorStop(0, '#000000');
   613→    voidGrad.addColorStop(0.7, '#000000');
   614→    voidGrad.addColorStop(0.92, isCritical ? '#660022' : glowColor + '88');
   615→    voidGrad.addColorStop(1, 'rgba(0,0,0,0)');
   616→    ctx.fillStyle = voidGrad;
   617→    ctx.fill();

   620→    ctx.beginPath();
   621→    ctx.arc(0, 0, rimR, 0, TAU);
   622→    ctx.strokeStyle = rimColor;
   623→    ctx.lineWidth = 3;
   624→    ctx.shadowBlur = 20;

   626→    ctx.stroke();

   630→    // Optimized: single beginPath for all arcs of the same style (where possible),
   631→    // single shadowBlur set, precompute radii once.
   632→    if (isDamaged) {
   633→      ctx.save();
   634→      ctx.globalCompositeOperation = 'lighter';
   635→      ctx.shadowBlur = 10;
   636→      ctx.shadowColor = rimGlowColor;
   637→      const arcCount = 3 + phase;
   638→      ctx.strokeStyle = isCritical ? 'rgba(255, 80, 120, 0.7)' : 'rgba(200, 120, 255, 0.6)';
   639→      ctx.lineWidth = 2;
   640→      // Precompute arc parameters
   641→      const baseRot = rotation * 4;
   642→      const t180 = time / 180;
   643→      const t250 = time / 250;
   644→      for (let i = 0; i < arcCount; i++) {
   645→        const aa = baseRot + (TAU / arcCount) * i;
   646→        const arcLen = Math.PI * (0.3 + Math.sin(t180 + i) * 0.15);
   647→        const arcRad = radius * (0.55 + Math.sin(t250 + i) * 0.1);
   648→        ctx.beginPath();
   649→        ctx.arc(0, 0, arcRad, aa, aa + arcLen);
   650→        ctx.stroke();
   651→      }
   652→      ctx.shadowBlur = 0;
   653→      ctx.restore();
   654→    }
   655→
   656→    // Layer 7 — Core singularity (pure black, absorbs light)
   657→    ctx.beginPath();
   658→    ctx.arc(0, 0, coreR, 0, TAU);
   659→    ctx.fillStyle = '#000000';
   660→    ctx.fill();
   661→
   662→    // Layer 8 — Phase warning glow (pulses red when boss is near death)
   663→    if (isCritical) {
   664→      const warnAlpha = (Math.sin(time / 150) + 1) / 2 * 0.4;
   665→      ctx.beginPath();
   666→      ctx.arc(0, 0, radius * 1.4, 0, TAU);
   667→      ctx.fillStyle = `rgba(255, 0, 51, ${warnAlpha})`;
   668→      ctx.globalCompositeOperation = 'lighter';
   669→      ctx.fill();
   670→      ctx.globalCompositeOperation = 'source-over';
   671→    }
   672→
   673→    // Flash effect (when hit)
   674→    if (flashTimer > 0) {
   675→      ctx.beginPath();
   676→      ctx.arc(0, 0, radius * 1.3, 0, TAU);
   677→      ctx.fillStyle = `rgba(255, 255, 255, ${flashAlpha * 0.85})`;
   678→      ctx.fill();
   679→    }
   680→
   681→    ctx.restore();
   682→    return;
   683→  }

   687→  }

   689→  const drawPart = (pathFn: () => void, partColor: string = color, scale: number = 1, isCore: boolean = false) => {
   690→    ctx.save();
   691→    ctx.beginPath();
   692→    pathFn();
   693→    
   694→    if (flashTimer > 0) {
   695→      ctx.fillStyle = `rgba(255,255,255,${flashAlpha * 0.9})`;
   696→      ctx.fill();
   697→      ctx.strokeStyle = '#ffffff';
   698→      ctx.lineWidth = isCore ? 2 : 4;
   699→      ctx.stroke();
   700→    } else {
   701→      if (isCore) {
   702→        // Cached time (shared across all drawPart calls for this enemy)
   703→        const now = Date.now();
   704→        const pulse = Math.sin(rotation * 12) * 0.15 + 0.85; 
   705→        const coreRadius = radius * scale;
   706→        const TAU = Math.PI * 2;
   707→
   708→        // 1. Energy Aura — запечённый спрайт (идентичен старому градиенту,
   709→        // но без createRadialGradient в каждом кадре)
   710→        ctx.save();
   711→        ctx.globalCompositeOperation = 'lighter';
   712→        const auraR = coreRadius * 3 * pulse;
   713→        ctx.drawImage(getAuraSprite(color), -auraR, -auraR, auraR * 2, auraR * 2);
   714→        ctx.restore();
   715→
   716→        // 1.5. Radiating sparks — batched: glow pass + core pass (2 fillStyle changes total)
   717→        ctx.save();
   718→        ctx.globalCompositeOperation = 'lighter';
   719→        const now100 = now / 100;
   720→        // Precompute spark positions
   721→        const sparkPositions: { x: number; y: number }[] = [];
   722→        for (let i = 0; i < 5; i++) {
   723→          const sparkAngle = rotation * 2 + (TAU / 5) * i;
   724→          const sparkDist = coreRadius * 1.5 + Math.sin(now100 + i) * coreRadius;
   725→          sparkPositions.push({ x: Math.cos(sparkAngle) * sparkDist, y: Math.sin(sparkAngle) * sparkDist });
   726→        }
   727→        // Glow pass (single fillStyle)
   728→        ctx.fillStyle = color + '66';
   729→        for (let i = 0; i < 5; i++) {
   730→          ctx.beginPath();
   731→          ctx.arc(sparkPositions[i].x, sparkPositions[i].y, 4 * pulse, 0, TAU);
   732→          ctx.fill();
   733→        }
   734→        // Core pass (single fillStyle)
   735→        ctx.fillStyle = '#ffffff';
   736→        for (let i = 0; i < 5; i++) {
   737→          ctx.beginPath();
   738→          ctx.arc(sparkPositions[i].x, sparkPositions[i].y, 1.5 * pulse, 0, TAU);
   739→          ctx.fill();
   740→        }
   741→        ctx.restore();
   742→
   743→        // 2. Swirling Vortex — reduced from 3 layers to 2 for performance,

   749→        const sin300 = Math.sin(now300);
   750→        const cos300 = Math.cos(now300);
   751→        for (let layer = 0; layer < 2; layer++) {
   752→          ctx.beginPath();
   753→          const numArms = 3 + layer; 
   754→          const layerRadius = coreRadius * (3.0 - layer * 0.5);
   755→          const offset = rotation * (layer * 0.4);

   760→            ctx.bezierCurveTo(
   761→              Math.cos(a) * layerRadius * 0.3, Math.sin(a) * layerRadius * 0.3,
   762→              Math.cos(a + 1.2) * layerRadius * 0.7, Math.sin(a + 1.2) * layerRadius * 0.7,
   763→              Math.cos(a + 2.0) * sin300 * layerRadius, Math.sin(a + 2.0) * cos300 * layerRadius
   764→            );
   765→          }
   766→          
   767→          ctx.strokeStyle = layer === 1 ? '#ffffff' : color;
   768→          ctx.globalAlpha = layer === 1 ? 0.2 * pulse : 0.27 * pulse;
   769→          ctx.lineWidth = (12 - layer * 3) * pulse;
   770→          ctx.lineCap = 'round';
   771→          ctx.stroke();
   772→          // Fake glow
   773→          ctx.lineWidth = (12 - layer * 3) * pulse * 2.5;
   774→          ctx.globalAlpha = ctx.globalAlpha * 0.3;
   775→          ctx.stroke();
   776→        }
   777→        ctx.restore();
   778→
   779→        // 3. Main Core Shape — запечённый спрайт (градиент + glow-обводка +
   780→        // белая обводка за один drawImage; рисуется в натуральном размере,
   781→        // поэтому ширины обводок совпадают с оригиналом)
   782→        const coreSpr = getShapeSprite(shape as string, color, coreRadius);
   783→        const halfSpr = coreSpr.width / 2;
   784→        ctx.drawImage(coreSpr, -halfSpr, -halfSpr);
   785→

   788→        ctx.rotate(-rotation * 5);

   790→        ctx.beginPath();

   798→        
   799→        // 5. Center Spark
   800→        ctx.beginPath();
   801→        ctx.arc(0, 0, coreRadius * 0.2, 0, TAU);
   802→        ctx.strokeStyle = 'rgba(255,255,255,0.4)';
   803→        ctx.lineWidth = 5;
   804→        ctx.stroke();
   805→        ctx.fillStyle = '#ffffff';
   806→        ctx.fill();
   807→      } else {
   808→        // Dynamic Neon Wireframe style — cached Date.now()
   809→        const now = Date.now();
   810→        const TAU = Math.PI * 2;
   811→        ctx.fillStyle = partColor + '15';
   812→        ctx.fill();
   813→
   814→        // Fake outer glow
   815→        const pulseGlow = Math.sin(now / 200) * 5 + 10;
   816→        ctx.strokeStyle = partColor + '44';
   817→        ctx.lineWidth = pulseGlow;
   818→        ctx.stroke();
   819→
   820→        ctx.strokeStyle = partColor;
   821→        ctx.lineWidth = 3;
   822→        ctx.stroke();

   824→        // Inner geometric floating lines
   825→        ctx.save();
   826→        ctx.rotate(rotation * 0.5);
   827→        ctx.beginPath();
   828→        ctx.scale(0.8, 0.8);
   829→        pathFn();
   830→        ctx.strokeStyle = partColor + '88';
   831→        ctx.setLineDash(DASH_ENEMY_INNER);
   832→        ctx.lineWidth = 2;
   833→        ctx.stroke();
   834→        ctx.setLineDash([]);
   835→        ctx.restore();
   836→
   837→        // Third layer: spinning arcs inside
   838→        ctx.save();
   839→        ctx.rotate(rotation * 1.5);
   840→        ctx.beginPath();
   841→        ctx.arc(0, 0, radius * 0.7, 0, Math.PI);
   842→        ctx.strokeStyle = partColor + 'dd';
   843→        ctx.lineWidth = 1;
   844→        ctx.stroke();
   845→        ctx.beginPath();
   846→        ctx.arc(0, 0, radius * 0.5, Math.PI, Math.PI * 2);
   847→        ctx.strokeStyle = '#ffffff88';
   848→        ctx.lineWidth = 1.5;
   849→        ctx.stroke();
   850→        ctx.restore();
   851→
   852→        // Second inner layer rotating in opposite direction
   853→        ctx.save();
   854→        ctx.rotate(-rotation * 0.8);
   855→        ctx.beginPath();
   856→        ctx.scale(0.6, 0.6);
   857→        pathFn();
   858→        ctx.strokeStyle = '#ffffffaa';
   859→        ctx.lineJoin = 'round';
   860→        ctx.lineWidth = 1.5;
   861→        ctx.stroke();
   862→        ctx.restore();
   863→
   864→        // Glass reflection — запечённый спрайт (клип по внутренней фигуре,
   865→        // как в оригинальной логике путей)
   866→        ctx.save();
   867→        ctx.save();
   868→        ctx.rotate(-rotation * 0.8);
   869→        ctx.scale(0.6, 0.6);
   870→        ctx.beginPath();
   871→        tracePath(ctx, shape as string, radius);
   872→        ctx.restore();
   873→        ctx.clip();
   874→        ctx.drawImage(
   875→          getSheenSprite(),
   876→          -radius * scale, -radius * scale,
   877→          radius * 2 * scale, radius * 2 * scale,
   878→        );
   879→        ctx.restore();
   880→      }
   881→    }
   882→    ctx.restore();
