    41→// Таблица hex-прозрачности (0-255 → "00".."ff") — раньше строилась
    42→// заново при каждом вызове renderParticles
    43→const ALPHA_HEX: string[] = new Array(256);

    48→// Кэш позиций звёзд — детерминированы индексом, пересчитываются

    50→const STAR_COUNT = 150;
    51→const starOrigX = new Float32Array(STAR_COUNT);

    54→let starCacheW = -1;

    57→function buildStarCache(W: number, H: number) {
    58→  for (let i = 0; i < STAR_COUNT; i++) {
    59→    starOrigX[i] = (Math.sin(i * 123.45) * 0.5 + 0.5) * W;
    60→    starOrigY[i] = (Math.cos(i * 678.90) * 0.5 + 0.5) * H;
    61→    starSpeed[i] = ((i % 3) + 1) * 0.15;
    62→  }
    63→  starCacheW = W;
    64→  starCacheH = H;
    65→}
    66→
    67→// Запечённые спрайты туманностей — раньше 3 createRadialGradient +
    68→// огромные blended-заливки каждый кадр, теперь 3 drawImage
    69→const NEBULA_SPRITES = [
    70→  bakeRadialSprite('bg:nebula:0', [
    71→    [0, 'rgba(100, 0, 255, 0.07)'],
    72→    [1, 'rgba(100, 0, 255, 0)'],
    73→  ], 256),
    74→  bakeRadialSprite('bg:nebula:1', [

    84→// Кэш градиента глубокого космоса (меняется только при resize)

    87→export function renderBackground(ctx: CanvasRenderingContext2D, W: number, H: number, frame: number, cx: number, cy: number) {
    88→  ctx.clearRect(0, 0, W, H);
    89→
    90→  // Deep space gradient — переиспользуем объект градиента между кадрами
    91→  if (!bgGradCache || bgGradCache.w !== W || bgGradCache.h !== H || bgGradCache.cx !== cx || bgGradCache.cy !== cy) {
    92→    const grad = ctx.createRadialGradient(cx, cy, 0, cx, cy, Math.max(W, H));
    93→    grad.addColorStop(0, '#0a0520');
    94→    grad.addColorStop(1, '#020108');
    95→    bgGradCache = { w: W, h: H, cx, cy, grad };
    96→  }
    97→  ctx.fillStyle = bgGradCache.grad;
    98→  ctx.fillRect(0, 0, W, H);
    99→
   100→  // Nebulae (Gas clouds) — запечённые спрайты вместо живых градиентов
   101→  ctx.globalCompositeOperation = 'screen';
   102→  const t = frame * 0.002;
   103→  const drawNebula = (nx: number, ny: number, r: number, sprite: HTMLCanvasElement) => {
   104→    ctx.drawImage(sprite, nx - r, ny - r, r * 2, r * 2);
   105→  };
   106→  drawNebula(W * 0.2 + Math.sin(t) * 150, H * 0.3 + Math.cos(t * 0.8) * 150, Math.max(W, H) * 0.5, NEBULA_SPRITES[0]);
   107→  drawNebula(W * 0.8 + Math.cos(t * 1.2) * 150, H * 0.7 + Math.sin(t * 1.1) * 150, Math.max(W, H) * 0.4, NEBULA_SPRITES[1]);
   108→  drawNebula(W * 0.5 + Math.sin(t * 0.5) * 200, H * 0.9 + Math.cos(t * 0.7) * 100, Math.max(W, H) * 0.4, NEBULA_SPRITES[2]);
   109→  ctx.globalCompositeOperation = 'source-over';
   110→
   111→  // Parallax Stars — позиции из кэша; группировка по слоям, чтобы
   112→  // fillStyle менялся 3 раза за кадр, а не 150
   113→  if (W !== starCacheW || H !== starCacheH) buildStarCache(W, H);
   114→  const starStyles = [
   115→    'rgba(100, 100, 150, 0.3)',  // far
   116→    'rgba(150, 200, 255, 0.6)',  // mid
   117→    'rgba(255, 255, 255, 0.9)',  // close
   118→  ];
   119→  for (let layer = 0; layer < 3; layer++) {
   120→    ctx.fillStyle = starStyles[layer];
   121→    for (let i = layer; i < STAR_COUNT; i += 3) {
   122→      const speed = starSpeed[i];
   123→      const twinkle = Math.sin(frame * 0.03 + i) * 0.4 + 0.6;
   124→      const size = (layer === 2 ? 1.5 : 0.8) * twinkle;
   125→
   126→      // Корректный wrap (звёзды больше не исчезают со временем)
   127→      const x = (((starOrigX[i] - frame * speed) % W) + W) % W;
   128→      const y = (((starOrigY[i] + frame * speed * 0.3) % H) + H) % H;
   129→
   130→      ctx.beginPath();
   131→      ctx.arc(x, y, size, 0, Math.PI * 2);
   132→      ctx.fill();
   133→
   134→      // Лёгкое свечение близких звёзд
   135→      if (layer === 2 && Math.sin(frame * 0.05 + i) > 0.8) {
   136→        ctx.beginPath();
   137→        ctx.arc(x, y, size * 2.5, 0, Math.PI * 2);
   138→        ctx.fillStyle = 'rgba(255, 255, 255, 0.4)';
   139→        ctx.fill();
   140→        ctx.fillStyle = starStyles[layer];
   141→      }
   142→    }
   143→  }
   144→
   145→  // Grid with perspective-like pulse (keep it subtle for the neon arcade feel)
   146→  ctx.strokeStyle = 'rgba(0,255,255,0.03)';
   147→  ctx.lineWidth = 1;
   148→  const gridSize = 80;
   149→  const offsetX = (frame * 0.2) % gridSize;
   150→  const offsetY = (frame * 0.2) % gridSize;
   151→
   152→  ctx.beginPath();
   153→  for (let gx = offsetX; gx < W; gx += gridSize) {
   154→    ctx.moveTo(gx, 0); ctx.lineTo(gx, H);
   155→  }
   156→  for (let gy = offsetY; gy < H; gy += gridSize) {
   157→    ctx.moveTo(0, gy); ctx.lineTo(W, gy);
   158→  }
   159→  ctx.stroke();
   160→
   161→  // Center rings with glow
   162→  for (let r = 80; r < Math.max(W, H); r += 150) {
   163→    ctx.beginPath();
   164→    ctx.arc(cx, cy, r, 0, Math.PI * 2);
   165→    const alpha = 0.02 + 0.02 * Math.sin(frame * 0.015 + r * 0.001);
   166→    ctx.strokeStyle = `rgba(0,255,255,${alpha})`;
   167→    ctx.lineWidth = 2;
   168→    ctx.stroke();
   169→  }
   170→}

   172→export function renderParticles(ctx: CanvasRenderingContext2D, particles: Particle[]) {
   173→  const TAU = Math.PI * 2;
   174→  // ALPHA_HEX — модульная таблица (создаётся один раз при загрузке модуля)
   175→
   176→  for (let i = 0; i < particles.length; i++) {
   177→    const p = particles[i];
   178→    const alpha = p.life / p.maxLife;
   179→    const r = p.radius * alpha;
   180→
   181→    // Fake glow
   182→    const glowAlphaByte = Math.min(255, Math.floor(alpha * 0.3 * 255));
   183→    ctx.beginPath();
   184→    ctx.arc(p.x, p.y, r * 2.5, 0, TAU);
   185→    ctx.fillStyle = p.color + ALPHA_HEX[glowAlphaByte];
   186→    ctx.fill();
   187→
   188→    // Core
   189→    const coreAlphaByte = Math.min(255, Math.floor(alpha * 255));
   190→    ctx.beginPath();
   191→    ctx.arc(p.x, p.y, r, 0, TAU);
   192→    ctx.fillStyle = p.color + ALPHA_HEX[coreAlphaByte];
   193→    ctx.fill();
   194→  }
   195→}

   197→export function renderShockwaves(ctx: CanvasRenderingContext2D, shockwaves: { x: number; y: number; radius: number; life: number; maxLife: number }[]) {
   198→  shockwaves.forEach(sw => {
   199→    const alpha = sw.life / sw.maxLife;
   200→    const t = 1 - alpha; // 0 at start, 1 at end
   201→
   202→    // Outer glow ring (wide, soft) — blue
   203→    ctx.beginPath();
   204→    ctx.arc(sw.x, sw.y, sw.radius, 0, Math.PI * 2);
   205→    ctx.strokeStyle = `rgba(0, 170, 255, ${alpha * 0.25})`;
   206→    ctx.lineWidth = 18;
   207→    ctx.stroke();
   208→
   209→    // Mid ring (main wave color) — blue
   210→    ctx.beginPath();
   211→    ctx.arc(sw.x, sw.y, sw.radius, 0, Math.PI * 2);
   212→    ctx.strokeStyle = `rgba(0, 170, 255, ${alpha * 0.85})`;
   213→    ctx.lineWidth = 5;
   214→    ctx.stroke();
   215→
   216→    // Bright inner edge
   217→    ctx.beginPath();
   218→    ctx.arc(sw.x, sw.y, sw.radius, 0, Math.PI * 2);
   219→    ctx.strokeStyle = `rgba(255, 255, 255, ${alpha * 0.6})`;
   220→    ctx.lineWidth = 2;
   221→    ctx.stroke();
   222→
   223→    // Trailing inner ring (slightly smaller, fading) — blue
   224→    if (t > 0.1) {
   225→      ctx.beginPath();
   226→      ctx.arc(sw.x, sw.y, sw.radius * 0.82, 0, Math.PI * 2);
   227→      ctx.strokeStyle = `rgba(0, 170, 255, ${alpha * 0.4})`;
   228→      ctx.lineWidth = 3;
   229→      ctx.stroke();
   230→    }
   231→  });
   232→}

   234→export function renderDustParticles(ctx: CanvasRenderingContext2D, dust: { x: number; y: number; radius: number; alpha: number; absorbed: boolean; absorbProgress: number }[]) {
   235→  // Very faint ambient dust — use 'lighter' for a subtle glow blend.
   236→  ctx.save();
   237→  ctx.globalCompositeOperation = 'lighter';
   238→  for (let i = 0; i < dust.length; i++) {
   239→    const d = dust[i];
   240→    if (d.alpha <= 0.01) continue;
   241→    const r = d.radius * (d.absorbed ? (1 - d.absorbProgress * 0.5) : 1);
   242→    // soft glow
   243→    ctx.beginPath();
   244→    ctx.arc(d.x, d.y, r * 2.2, 0, Math.PI * 2);
   245→    ctx.fillStyle = `rgba(180, 200, 255, ${d.alpha * 0.3})`;
   246→    ctx.fill();
   247→    // bright core
   248→    ctx.beginPath();
   249→    ctx.arc(d.x, d.y, r, 0, Math.PI * 2);
   250→    ctx.fillStyle = `rgba(220, 230, 255, ${d.alpha})`;
   251→    ctx.fill();
   252→  }
   253→  ctx.restore();
   254→}

   256→export function renderGravityBeams(ctx: CanvasRenderingContext2D, beams: { x1: number; y1: number; x2: number; y2: number; life: number; maxLife: number }[]) {
   257→  // Purple animated energy beams from the boss to enemies.
   258→  // Multiple layers for a volumetric look.
   259→  const TAU = Math.PI * 2;
   260→  const time = Date.now();
   261→
   262→  ctx.save();
   263→  ctx.globalCompositeOperation = 'lighter';
   264→
   265→  for (let i = 0; i < beams.length; i++) {
   266→    const beam = beams[i];
   267→    const alpha = beam.life / beam.maxLife;
   268→    const dx = beam.x2 - beam.x1;
   269→    const dy = beam.y2 - beam.y1;
   270→    const len = Math.sqrt(dx * dx + dy * dy);
   271→    if (len < 1) continue;
   272→    const angle = Math.atan2(dy, dx);
   273→
   274→    ctx.save();
   275→    ctx.translate(beam.x1, beam.y1);
   276→    ctx.rotate(angle);
   277→
   278→    // Layer 1: Wide outer glow
   279→    ctx.beginPath();
   280→    ctx.moveTo(0, 0);
   281→    ctx.lineTo(len, 0);
   282→    ctx.strokeStyle = `rgba(191, 0, 255, ${alpha * 0.15})`;
   283→    ctx.lineWidth = 16;
   284→    ctx.lineCap = 'round';
   285→    ctx.stroke();
   286→
   287→    // Layer 2: Mid beam
   288→    ctx.beginPath();
   289→    ctx.moveTo(0, 0);
   290→    ctx.lineTo(len, 0);
   291→    ctx.strokeStyle = `rgba(191, 0, 255, ${alpha * 0.4})`;
   292→    ctx.lineWidth = 6;
   293→    ctx.stroke();
   294→
   295→    // Layer 3: Animated energy pulses traveling along the beam
   296→    const pulseCount = Math.max(2, Math.floor(len / 40));
   297→    for (let p = 0; p < pulseCount; p++) {
   298→      const phase = ((time / 200) + p / pulseCount) % 1;
   299→      const px = phase * len;
   300→      const pulseR = 4 + Math.sin(time / 100 + p) * 2;
   301→      // glow
   302→      ctx.beginPath();
   303→      ctx.arc(px, 0, pulseR * 2, 0, TAU);
   304→      ctx.fillStyle = `rgba(191, 0, 255, ${alpha * 0.3})`;
   305→      ctx.fill();
   306→      // core
   307→      ctx.beginPath();
   308→      ctx.arc(px, 0, pulseR, 0, TAU);
   309→      ctx.fillStyle = `rgba(255, 255, 255, ${alpha * 0.8})`;
   310→      ctx.fill();
   311→    }
   312→
   313→    // Layer 4: Bright core line
   314→    ctx.beginPath();
   315→    ctx.moveTo(0, 0);
   316→    ctx.lineTo(len, 0);
   317→    ctx.strokeStyle = `rgba(255, 255, 255, ${alpha * 0.6})`;
   318→    ctx.lineWidth = 1.5;
   319→    ctx.stroke();
   320→
   321→    ctx.restore();
   322→  }
   323→
   324→  ctx.restore();
   325→}

   327→export function renderMiniBlackHoles(ctx: CanvasRenderingContext2D, blackHoles: { x: number; y: number; radius: number; absorbRadius: number; life: number; maxLife: number; angle: number }[]) {
   328→  // Redesigned mini black holes — detailed black-white-purple cosmic voids.
   329→  // No rotating rings (unlike the boss) — instead: swirling particle streams,
   330→  // pulsing event horizon, energy tendrils, and a shimmering accretion disk.
   331→  const TAU = Math.PI * 2;
   332→  const time = Date.now();
   333→
   334→  for (let i = 0; i < blackHoles.length; i++) {
   335→    const bh = blackHoles[i];
   336→    const r = bh.radius;
   337→
   338→    // Fade in/out
   339→    let alpha = 1;
   340→    if (bh.life < 60) alpha = bh.life / 60;
   341→    if (bh.life > bh.maxLife - 30) alpha = (bh.maxLife - bh.life) / 30;
   342→
   343→    // Pulsing factor — subtle breathing
   344→    const pulse = 1 + Math.sin(time / 300 + i) * 0.06;
   345→
   346→    ctx.save();
   347→    ctx.translate(bh.x, bh.y);
   348→
   349→    // === Layer 0: Outer absorb field — swirling distortion ===
   350→    // Запечённый спрайт (раньше — createRadialGradient каждый кадр)
   351→    ctx.save();
   352→    ctx.globalCompositeOperation = 'lighter';
   353→    const fieldR = bh.absorbRadius * pulse;
   354→    ctx.globalAlpha = alpha;
   355→    ctx.drawImage(getBHFieldSprite(), -fieldR, -fieldR, fieldR * 2, fieldR * 2);
   356→    ctx.restore();
   357→
   358→    // === Layer 1: Swirling particle stream — dots spiraling inward ===
   359→    ctx.save();
   360→    ctx.globalCompositeOperation = 'lighter';
   361→    ctx.rotate(bh.angle * 0.5);
   362→    const particleCount = 14;
   363→    for (let p = 0; p < particleCount; p++) {
   364→      const spiralAngle = bh.angle * 2 + (TAU / particleCount) * p;
   365→      const spiralT = (p / particleCount + Math.sin(time / 400 + p) * 0.05) % 1;
   366→      const spiralR = r * 1.0 + (bh.absorbRadius - r) * (1 - spiralT);
   367→      const sx = Math.cos(spiralAngle) * spiralR;
   368→      const sy = Math.sin(spiralAngle) * spiralR;
   369→      const pAlpha = alpha * spiralT * 0.6;
   370→      const pSize = 1 + spiralT * 2;
   371→      // glow
   372→      ctx.beginPath();
   373→      ctx.arc(sx, sy, pSize * 2, 0, TAU);
   374→      ctx.fillStyle = `rgba(180, 120, 255, ${pAlpha * 0.3})`;
   375→      ctx.fill();
   376→      // core
   377→      ctx.beginPath();
   378→      ctx.arc(sx, sy, pSize, 0, TAU);
   379→      ctx.fillStyle = `rgba(220, 200, 255, ${pAlpha})`;
   380→      ctx.fill();
   381→    }
   382→    ctx.restore();
   383→
   384→    // === Layer 2: Energy tendrils — short jagged arcs around the void ===
   385→    ctx.save();
   386→    ctx.globalCompositeOperation = 'lighter';
   387→    ctx.rotate(-bh.angle * 1.5);
   388→    const tendrilCount = 5;
   389→    for (let t = 0; t < tendrilCount; t++) {
   390→      const baseAngle = (TAU / tendrilCount) * t + Math.sin(time / 250 + t) * 0.2;
   391→      const tendrilLen = 0.6 + Math.sin(time / 180 + t * 1.5) * 0.15;
   392→      ctx.beginPath();
   393→      ctx.arc(0, 0, r * 1.3, baseAngle, baseAngle + tendrilLen);
   394→      ctx.strokeStyle = `rgba(160, 100, 220, ${alpha * 0.45})`;
   395→      ctx.lineWidth = 3;
   396→      ctx.lineCap = 'round';
   397→      ctx.stroke();
   398→      // bright inner edge
   399→      ctx.beginPath();
   400→      ctx.arc(0, 0, r * 1.3, baseAngle, baseAngle + tendrilLen * 0.6);
   401→      ctx.strokeStyle = `rgba(255, 240, 255, ${alpha * 0.3})`;
   402→      ctx.lineWidth = 1;
   403→      ctx.stroke();
   404→    }
   405→    ctx.restore();
   406→
   407→    // === Layer 3: Accretion disk — thin luminous elliptical band ===
   408→    // (In the old code the disk gradient almost did not read —
   409→    // now it is a neat ring sprite with soft edges)
   410→    ctx.save();
   411→    ctx.rotate(bh.angle * 0.8);
   412→    ctx.scale(1, 0.25);
   413→    ctx.globalAlpha = alpha;
   414→    const diskR = r * 2.0 * pulse;
   415→    ctx.drawImage(getBHDiskSprite(), -diskR, -diskR, diskR * 2, diskR * 2);
   416→    ctx.restore();
   417→
   418→    // === Layer 4: Event horizon — dark void with layered gradient ===
   419→    // Baked sprite (globalAlpha handles fade in/out)
   420→    ctx.save();
   421→    ctx.globalAlpha = alpha;
   422→    const voidR = r * pulse;
   423→    ctx.drawImage(getBHVoidSprite(), -voidR, -voidR, voidR * 2, voidR * 2);
   424→    ctx.restore();
   425→
   426→    // === Layer 5: Photon ring — thin bright ring at the edge of the void ===
   427→    ctx.beginPath();
   428→    ctx.arc(0, 0, r * 0.95 * pulse, 0, TAU);
   429→    ctx.strokeStyle = `rgba(255, 240, 255, ${alpha * 0.5})`;
   430→    ctx.lineWidth = 1;
   431→    ctx.stroke();
   432→
   433→    // === Layer 6: Pulsing inner core — small bright dot that flickers ===
   434→    const coreFlicker = 0.6 + Math.sin(time / 80 + i * 2) * 0.3;
   435→    ctx.save();
   436→    ctx.globalCompositeOperation = 'lighter';
   437→    ctx.globalAlpha = Math.min(1, alpha * coreFlicker);
   438→    const coreR = r * 0.3 * pulse;
   439→    ctx.drawImage(getBHCoreSprite(), -coreR, -coreR, coreR * 2, coreR * 2);
   440→    ctx.restore();
   441→
   442→    // === Layer 7: Inner pure black singularity ===
   443→    ctx.beginPath();
   444→    ctx.arc(0, 0, r * 0.25 * pulse, 0, TAU);
   445→    ctx.fillStyle = '#000000';
   446→    ctx.fill();
   447→
   448→    // === Layer 8: Spawn/despawn flash ===
   449→    if (bh.life > bh.maxLife - 20) {
   450→      const flashAlpha = (bh.maxLife - bh.life) / 20;
   451→      ctx.save();
   452→      ctx.globalCompositeOperation = 'lighter';
   453→      ctx.beginPath();
   454→      ctx.arc(0, 0, r * 2.5 * (1 - flashAlpha), 0, TAU);
   455→      ctx.strokeStyle = `rgba(200, 150, 255, ${flashAlpha * 0.6})`;
   456→      ctx.lineWidth = 3;
   457→      ctx.stroke();
   458→      ctx.restore();
   459→    }
   460→
   461→    ctx.restore();
   462→  }

   465→export function renderProjectiles(ctx: CanvasRenderingContext2D, projectiles: Projectile[]) {
   466→  projectiles.forEach(p => {
   467→    // Magnet arrows use a magenta/purple palette, crit arrows use orange, regular use cyan
   468→    const isMagnet = p.isMagnet;
   469→    const baseR = isMagnet ? 255 : (p.isCrit ? 255 : 0);
   470→    const baseG = isMagnet ? 0 : (p.isCrit ? 170 : 255);
   471→    const baseB = isMagnet ? 255 : (p.isCrit ? 0 : 255);
   472→
   473→    // Draw trail
   474→    if (p.trail) {
   475→      p.trail.forEach((t, ti) => {
   476→        const alpha = (ti / p.trail.length) * 0.6;
   477→        const r = p.radius * (ti / p.trail.length) * (isMagnet ? 1.1 : 0.8);
   478→        ctx.beginPath();
   479→        ctx.arc(t.x, t.y, r, 0, Math.PI * 2);
   480→        ctx.fillStyle = `rgba(${baseR}, ${baseG}, ${baseB}, ${alpha})`;
   481→        ctx.fill();
   482→        // Magnet trail gets an extra white-hot core
   483→        if (isMagnet) {
   484→          ctx.beginPath();
   485→          ctx.arc(t.x, t.y, r * 0.4, 0, Math.PI * 2);
   486→          ctx.fillStyle = `rgba(255, 255, 255, ${alpha * 0.7})`;
   487→          ctx.fill();
   488→        }
   489→      });
   490→    }
   491→
   492→    // Magnet aura — pulsing magnetic field rings around the arrow (un-rotated)
   493→    if (isMagnet) {
   494→      const pulseT = Date.now() / 120;
   495→      for (let ring = 0; ring < 3; ring++) {
   496→        const phase = (pulseT + ring * 1.2) % 3;
   497→        const ringR = 8 + phase * 10;
   498→        const ringAlpha = Math.max(0, 0.5 - phase / 6);
   499→        ctx.beginPath();
   500→        ctx.arc(p.x, p.y, ringR, 0, Math.PI * 2);
   501→        ctx.strokeStyle = `rgba(255, 0, 255, ${ringAlpha})`;
   502→        ctx.lineWidth = 2;
   503→        ctx.stroke();
   504→      }
   505→      // Soft outer halo
   506→      const haloGrad = ctx.createRadialGradient(p.x, p.y, 0, p.x, p.y, 22);
   507→      haloGrad.addColorStop(0, 'rgba(255, 0, 255, 0.35)');
   508→      haloGrad.addColorStop(0.5, 'rgba(191, 0, 255, 0.15)');
   509→      haloGrad.addColorStop(1, 'rgba(191, 0, 255, 0)');
   510→      ctx.beginPath();
   511→      ctx.arc(p.x, p.y, 22, 0, Math.PI * 2);
   512→      ctx.fillStyle = haloGrad;
   513→      ctx.fill();
   514→    }
   515→
   516→    ctx.save();
   517→    ctx.translate(p.x, p.y);
   518→    ctx.rotate(Math.atan2(p.vy, p.vx));
   519→
   520→    const arrowLen = 30;
   521→
   522→    const glowColor = `rgba(${baseR}, ${baseG}, ${baseB}, ${isMagnet ? 0.7 : 0.5})`;
   523→    const shadowColor = isMagnet ? '#ff00ff' : (p.isCrit ? '#ffaa00' : '#00ffff');
   524→    const fillHeadEnd = isMagnet ? '#660066' : (p.isCrit ? '#aa5500' : '#005555');
   525→    const fletchColor = `rgba(${baseR}, ${baseG}, ${baseB}, 0.9)`;
   526→
   527→    // Arrow shaft glow (thicker for magnet)
   528→    ctx.beginPath();
   529→    ctx.moveTo(-25, 0);
   530→    ctx.lineTo(arrowLen, 0);
   531→    ctx.strokeStyle = `rgba(${baseR}, ${baseG}, ${baseB}, ${isMagnet ? 0.3 : 0.2})`;
   532→    ctx.lineWidth = isMagnet ? 16 : 12;
   533→    ctx.stroke();
   534→
   535→    ctx.beginPath();
   536→    ctx.moveTo(-25, 0);
   537→    ctx.lineTo(arrowLen, 0);
   538→    ctx.strokeStyle = glowColor;
   539→    ctx.lineWidth = isMagnet ? 5 : 4;
   540→    ctx.stroke();
   541→
   542→    // Arrow shaft core
   543→    ctx.beginPath();
   544→    ctx.moveTo(-25, 0);
   545→    ctx.lineTo(arrowLen, 0);
   546→    ctx.strokeStyle = '#ffffff';
   547→    ctx.lineWidth = 2;
   548→    ctx.stroke();
   549→
   550→    // High-tech Arrow head
   551→    ctx.beginPath();
   552→    ctx.moveTo(arrowLen + 8, 0);
   553→    ctx.lineTo(arrowLen - 6, -6);
   554→    ctx.lineTo(arrowLen - 2, 0);
   555→    ctx.lineTo(arrowLen - 6, 6);
   556→    ctx.closePath();
   557→    const headGrad = ctx.createLinearGradient(arrowLen - 6, -6, arrowLen + 8, 8);
   558→    headGrad.addColorStop(0, '#ffffff');
   559→    headGrad.addColorStop(0.4, shadowColor);
   560→    headGrad.addColorStop(1, fillHeadEnd);
   561→    ctx.fillStyle = headGrad;
   562→    ctx.fill();
   563→
   564→    // Arrow fletching (Energy fins)
   565→    ctx.beginPath();
   566→    ctx.moveTo(-25, 0);
   567→    ctx.lineTo(-33, -6);
   568→    ctx.lineTo(-28, 0);
   569→    ctx.lineTo(-33, 6);
   570→    ctx.closePath();
   571→    ctx.fillStyle = fletchColor;
   572→    ctx.fill();
   573→
   574→    // Extra energy rings around the arrow
   575→    ctx.beginPath();
   576→    ctx.ellipse(0, 0, 3, 9, 0, 0, Math.PI * 2);
   577→    ctx.strokeStyle = glowColor;
   578→    ctx.lineWidth = isMagnet ? 2 : 1.5;
   579→    ctx.stroke();
   580→
   581→    // Magnet arrows get an extra small magnet icon glow at the tip
   582→    if (isMagnet) {
   583→      ctx.beginPath();
   584→      ctx.arc(arrowLen + 4, 0, 5, 0, Math.PI * 2);
   585→      ctx.fillStyle = 'rgba(255, 255, 255, 0.8)';
   586→      ctx.fill();
   587→      ctx.beginPath();
   588→      ctx.arc(arrowLen + 4, 0, 8, 0, Math.PI * 2);
   589→      ctx.strokeStyle = 'rgba(255, 0, 255, 0.6)';
   590→      ctx.lineWidth = 1.5;
   591→      ctx.stroke();
   592→    }
   593→
   594→    ctx.restore();
   595→  });
   596→}

   598→export function renderEnemies(
   599→  ctx: CanvasRenderingContext2D,
   600→  enemies: Enemy[],
   601→  opts?: { bossName?: string; bossDanger?: string },
   602→) {
   603→  const time = Date.now(); // one time per frame for all enemies
   604→  enemies.forEach(e => {
   605→    const isGhost = e.id === 'ghost';
   606→    const isPhantom = e.id === 'phantom';
   607→    const hasAlpha = isGhost || isPhantom;
   608→    const alpha = hasAlpha ? (e.teleportAlpha ?? 1) : 1;
   609→    const ringAnim = e.teleportRingAnim ?? 0;
   610→
   611→    // Анимация появления после телепортации: расширяющееся кольцо (рисуем до globalAlpha)
   612→    if (isGhost && ringAnim > 0) {
   613→      const ringProgress = 1 - ringAnim / 30;
   614→      const ringRadius = e.radius + ringProgress * e.radius * 2.5;
   615→      const ringAlpha = ringAnim / 30;
   616→      ctx.globalAlpha = ringAlpha;
   617→      // Внешнее кольцо
   618→      ctx.beginPath();
   619→      ctx.arc(e.x, e.y, ringRadius, 0, Math.PI * 2);
   620→      ctx.strokeStyle = e.color + '44';
   621→      ctx.lineWidth = 15 * ringAlpha;
   622→      ctx.stroke();
   623→      ctx.strokeStyle = e.color;
   624→      ctx.lineWidth = 3 * ringAlpha;
   625→      ctx.stroke();
   626→      // Второе кольцо чуть меньше
   627→      ctx.beginPath();
   628→      ctx.arc(e.x, e.y, ringRadius * 0.65, 0, Math.PI * 2);
   629→      ctx.strokeStyle = 'rgba(255,255,255,0.4)';
   630→      ctx.lineWidth = 8 * ringAlpha;
   631→      ctx.stroke();
   632→      ctx.strokeStyle = '#ffffff';
   633→      ctx.lineWidth = 1.5 * ringAlpha;
   634→      ctx.stroke();
   635→    }
   636→
   637→    ctx.globalAlpha = alpha;
   638→
   639→    // Draw Enemy Trail — ONE polyline with a gradient (previously ~26
   640→    // stroke calls per enemy: 3 strokes per segment; now — 3 for the entire trail,
   641→    // and the alpha fade is smoother)
   642→    if (e.trail && e.trail.length > 1) {
   643→      ctx.save();
   644→      ctx.lineCap = 'round';
   645→      ctx.lineJoin = 'round';
   646→      const first = e.trail[0];               // oldest point
   647→      const last = e.trail[e.trail.length - 1]; // newest (near the enemy)
   648→
   649→      // Gradient: bright near the enemy → transparent towards the tail
   650→      const trailGrad = ctx.createLinearGradient(last.x, last.y, first.x, first.y);
   651→      trailGrad.addColorStop(0, e.color);
   652→      trailGrad.addColorStop(1, e.color + '00');
   653→
   654→      ctx.beginPath();
   655→      ctx.moveTo(first.x, first.y);
   656→      for (let ti = 1; ti < e.trail.length; ti++) {
   657→        ctx.lineTo(e.trail[ti].x, e.trail[ti].y);
   658→      }
   659→      ctx.lineTo(e.x, e.y);
   660→
   661→      // Soft wide band
   662→      ctx.strokeStyle = trailGrad;
   663→      ctx.globalAlpha = alpha * 0.3;
   664→      ctx.lineWidth = e.radius * 1.5;
   665→      ctx.stroke();
   666→      // Bright core of the trail
   667→      ctx.globalAlpha = alpha * 0.5;
   668→      ctx.lineWidth = e.radius * 0.8;
   669→      ctx.stroke();
   670→      // White-hot center
   671→      const whiteGrad = ctx.createLinearGradient(last.x, last.y, first.x, first.y);
   672→      whiteGrad.addColorStop(0, 'rgba(255,255,255,0.9)');
   673→      whiteGrad.addColorStop(1, 'rgba(255,255,255,0)');
   674→      ctx.strokeStyle = whiteGrad;
   675→      ctx.globalAlpha = alpha * 0.35;
   676→      ctx.lineWidth = Math.max(1, e.radius * 0.25);
   677→      ctx.stroke();
   678→
   679→      ctx.restore();
   680→      ctx.globalAlpha = alpha; // restore
   681→    }
   682→
   683→    // Pulse glow ring
   684→    ctx.save();
   685→    ctx.translate(e.x, e.y);
   686→    // Add dynamic rotation to the pulse ring
   687→    ctx.rotate((time / 1000) * (e.id === 'tank' ? 0.5 : 1.5));
   688→    ctx.beginPath();
   689→    ctx.arc(0, 0, e.pulseRadius + Math.sin(time / 150) * 2 + 4, 0, Math.PI * 2);
   690→    ctx.strokeStyle = e.color + 'aa';
   691→    ctx.lineWidth = 1.5;
   692→    ctx.setLineDash(DASH_PULSE_RING);
   693→    ctx.stroke();
   694→    // fake pulse glow
   695→    ctx.strokeStyle = e.color + '44';
   696→    ctx.lineWidth = 6;
   697→    ctx.stroke();
   698→    
   699→    // Additional inner dashed ring
   700→    ctx.beginPath();
   701→    ctx.rotate(-time / 500);
   702→    ctx.arc(0, 0, e.radius + 6, 0, Math.PI * 2);
   703→    ctx.strokeStyle = e.color + '66';
   704→    ctx.lineWidth = 1;
   705→    ctx.setLineDash(DASH_INNER_RING);
   706→    ctx.stroke();
   707→    ctx.setLineDash([]);
   708→    ctx.restore();
   709→
   710→    // Fade-in ring (пока появляется после телепорта)
   711→    if (isGhost && alpha < 1 && !e.teleportFading) {
   712→      ctx.beginPath();
   713→      ctx.arc(e.x, e.y, e.radius * (1.5 + (1 - alpha)), 0, Math.PI * 2);
   714→      ctx.strokeStyle = e.color + '66';
   715→      ctx.lineWidth = 10;
   716→      ctx.stroke();
   717→      ctx.strokeStyle = e.color;
   718→      ctx.lineWidth = 2;
   719→      ctx.stroke();
   720→    }
   721→
   722→    // Индикатор оставшихся телепортаций (маленькие точки над врагом)
   723→    if (isGhost && (e.teleportsLeft ?? 0) > 0 && alpha >= 1) {
   724→      ctx.globalAlpha = 0.8;
   725→      const dots = e.teleportsLeft ?? 0;
   726→      for (let d = 0; d < dots; d++) {
   727→        const dotX = e.x - (dots - 1) * 5 + d * 10;
   728→        const dotY = e.y - e.radius - 22;
   729→        ctx.beginPath();
   730→        ctx.arc(dotX, dotY, 3, 0, Math.PI * 2);
   731→        ctx.fillStyle = e.color;
   732→        ctx.shadowBlur = 8;
   733→        ctx.shadowColor = e.color;
   734→        ctx.fill();
   735→        ctx.shadowBlur = 0;
   736→      }
   737→      ctx.globalAlpha = alpha;
   738→    }
   739→
   740→    drawShape(ctx, e.shape, e.x, e.y, e.radius, e.color, e.glowColor, e.angle, e.flashTimer, e.hp, e.maxHp);
   741→
   742→    // HP bar -> Curved Arc HP Bar
   743→    if (e.maxHp > 1 && e.hp < e.maxHp) {
   744→      const radius = e.radius + 12; // place it just outside the enemy
   745→      const startAngle = -Math.PI; // Top left
   746→      const endAngle = 0; // Top right
   747→      const hpPercentage = Math.max(0, e.hp / e.maxHp);
   748→      
   749→      // Calculate the angle for the current HP
   750→      const currentAngle = startAngle + (endAngle - startAngle) * hpPercentage;
   751→
   752→      ctx.save();
   753→      ctx.translate(e.x, e.y);
   754→      
   755→      // Background track
   756→      ctx.beginPath();
   757→      ctx.arc(0, 0, radius, startAngle, endAngle);
   758→      ctx.strokeStyle = 'rgba(0,0,0,0.5)';
   759→      ctx.lineWidth = 3;
   760→      ctx.lineCap = 'round';
   761→      ctx.stroke();
   762→
   763→      // HP value
   764→      ctx.beginPath();
   765→      ctx.arc(0, 0, radius, startAngle, currentAngle);
   766→      ctx.strokeStyle = e.color + '66';
   767→      ctx.lineWidth = 9;
   768→      ctx.lineCap = 'round';
   769→      ctx.stroke();
   770→
   771→      ctx.beginPath();
   772→      ctx.arc(0, 0, radius, startAngle, currentAngle);
   773→      ctx.strokeStyle = e.color;
   774→      ctx.lineWidth = 3;
   775→      ctx.lineCap = 'round';
   776→      ctx.stroke();
   777→      
   778→      ctx.restore();
   779→    }
   780→    
   781→    // Epic Huge Boss HP
   782→    if (e.id === 'boss_omega') {
   783→      const W = ctx.canvas.width;
   784→      const bw = W * 0.6;
   785→      const bh = 22;
   786→      const bx = (W - bw) / 2;
   787→      const by = 40;
   788→
   789→      ctx.save();
   790→      const hpPercentage = Math.max(0, e.hp / e.maxHp);
   791→      // Phase: 0 (full) → 2 (near death)
   792→      const bossPhase = Math.min(2, Math.floor((1 - hpPercentage) * 3));
   793→      const barColor = bossPhase >= 2 ? '#ff0044' : (bossPhase >= 1 ? '#ff00aa' : e.glowColor);
   794→
   795→      // Outer frame glow
   796→      ctx.shadowBlur = 18;
   797→      ctx.shadowColor = barColor;
   798→      ctx.strokeStyle = barColor;
   799→      ctx.lineWidth = 2;
   800→      ctx.strokeRect(bx - 2, by - 2, bw + 4, bh + 4);
   801→
   802→      // bg track
   803→      ctx.shadowBlur = 0;
   804→      ctx.fillStyle = 'rgba(0,0,0,0.85)';
   805→      ctx.fillRect(bx, by, bw, bh);
   806→
   807→      // inner border
   808→      ctx.strokeStyle = 'rgba(255,255,255,0.15)';
   809→      ctx.lineWidth = 1;
   810→      ctx.strokeRect(bx, by, bw, bh);
   811→
   812→      // HP fill — gradient
   813→      const fillGrad = ctx.createLinearGradient(bx, by, bx, by + bh);
   814→      fillGrad.addColorStop(0, '#ffffff');
   815→      fillGrad.addColorStop(0.3, barColor);
   816→      fillGrad.addColorStop(1, bossPhase >= 2 ? '#660022' : '#330055');
   817→      ctx.fillStyle = fillGrad;
   818→      ctx.shadowBlur = 12;
   819→      ctx.shadowColor = barColor;
   820→      ctx.fillRect(bx, by, bw * hpPercentage, bh);
   821→
   822→      // Sheen overlay on the filled portion
   823→      if (hpPercentage > 0) {
   824→        ctx.shadowBlur = 0;
   825→        const sheenGrad = ctx.createLinearGradient(bx, by, bx, by + bh);
   826→        sheenGrad.addColorStop(0, 'rgba(255,255,255,0.4)');
   827→        sheenGrad.addColorStop(0.5, 'rgba(255,255,255,0.1)');
   828→        sheenGrad.addColorStop(1, 'rgba(255,255,255,0)');
   829→        ctx.fillStyle = sheenGrad;
   830→        ctx.fillRect(bx, by, bw * hpPercentage, bh);
   831→      }
   832→
   833→      // Phase tick marks (33% and 66%)
   834→      ctx.shadowBlur = 0;
   835→      ctx.strokeStyle = 'rgba(255,255,255,0.3)';
   836→      ctx.lineWidth = 1;
   837→      [0.33, 0.66].forEach(t => {
   838→        const tx = bx + bw * t;
   839→        ctx.beginPath();
   840→        ctx.moveTo(tx, by);
   841→        ctx.lineTo(tx, by + bh);
   842→        ctx.stroke();
   843→      });
   844→
   845→      // Title with glow
   846→      ctx.fillStyle = '#ffffff';
   847→      ctx.font = 'bold 18px Orbitron';
   848→      ctx.shadowBlur = 8;
   849→      ctx.shadowColor = barColor;
   850→      ctx.textAlign = 'center';
   851→      const bossLabel = (opts?.bossName ?? e.name).toUpperCase() + (bossPhase >= 2 ? ' — ' + (opts?.bossDanger ?? '') : '');
   852→      ctx.fillText(bossLabel, W / 2, by - 12);
   853→
   854→      // HP percentage text
   855→      ctx.font = 'bold 11px Orbitron';
   856→      ctx.shadowBlur = 4;
   857→      ctx.fillStyle = 'rgba(255,255,255,0.9)';
   858→      ctx.fillText(`${Math.round(hpPercentage * 100)}%`, W / 2, by + bh / 2 + 4);
   859→
   860→      ctx.restore();
   861→    }
   862→
   863→    ctx.globalAlpha = 1;
   864→  });
   865→}

   867→export function renderBow(
   868→  ctx: CanvasRenderingContext2D,
   869→  cx: number,
   870→  cy: number,
   871→  mx: number,
   872→  my: number,

   906→  ctx.arc(0, 0, 18 * pulse, 0, Math.PI * 2);

   917→  bowGrad.addColorStop(0, '#00ffff');
   918→  bowGrad.addColorStop(0.2, '#ffffff');
   919→  bowGrad.addColorStop(0.5, '#00aaff');
   920→  bowGrad.addColorStop(0.8, '#ffffff');
   921→  bowGrad.addColorStop(1, '#00ffff');
   922→
   923→  // Mechanical Bow Limbs
   924→  ctx.beginPath();
   925→  // Top limb
   926→  ctx.moveTo(-12 * pulse, -38 * pulse);
   927→  ctx.lineTo(8 * pulse, -25 * pulse);
   928→  ctx.lineTo(0, -15 * pulse);
   929→  // Bottom limb

   935→  ctx.lineWidth = 7;
   936→  ctx.lineCap = 'round';
   937→  ctx.lineJoin = 'round';
   938→  ctx.stroke();
   939→  
   940→  // Inner mechanical details
   941→  ctx.beginPath();
   942→  ctx.moveTo(-8 * pulse, -35 * pulse);
   943→  ctx.lineTo(5 * pulse, -25 * pulse);
   944→  ctx.moveTo(-8 * pulse, 35 * pulse);
   945→  ctx.lineTo(5 * pulse, 25 * pulse);
   946→  ctx.strokeStyle = 'rgba(255,255,255,0.8)';
   947→  ctx.lineWidth = 2;
   948→  ctx.stroke();
   949→
   950→  // Floating stabilizers
   951→  const time = Date.now() * 0.005;
   952→  const stabOffset = Math.sin(time) * 4;
   953→  ctx.beginPath();
   954→  ctx.moveTo(-18 * pulse, -45 * pulse + stabOffset);
   955→  ctx.lineTo(-8 * pulse, -45 * pulse + stabOffset);
   956→  ctx.moveTo(-18 * pulse, 45 * pulse - stabOffset);
   957→  ctx.lineTo(-8 * pulse, 45 * pulse - stabOffset);

   959→  ctx.lineWidth = 3;
   960→  ctx.lineCap = 'round';
   961→  ctx.stroke();
   962→
   963→  // Specular highlight on the core
   964→  ctx.beginPath();
   965→  ctx.ellipse(-6 * pulse, -6 * pulse, 8 * pulse, 4 * pulse, Math.PI / 4, 0, Math.PI * 2);
   966→  ctx.fillStyle = 'rgba(255,255,255,0.7)';
   967→  ctx.fill();
   968→
   969→  // Bow string (Energy tethers)
   970→  ctx.beginPath();
   971→  ctx.moveTo(-12 * pulse, -38 * pulse);
   972→  ctx.lineTo(15 * pulse, 0);
   973→  ctx.lineTo(-12 * pulse, 38 * pulse);
   974→  ctx.strokeStyle = 'rgba(0,255,255,0.6)';
   975→  ctx.lineWidth = 2;
   976→  ctx.stroke();
   977→  
   978→  ctx.beginPath();
   979→  ctx.moveTo(-12 * pulse, -38 * pulse);
   980→  ctx.lineTo(26 * pulse, 0);
   981→  ctx.lineTo(-12 * pulse, 38 * pulse);
   982→  ctx.strokeStyle = 'rgba(255,255,255,0.9)';
   983→  ctx.lineWidth = 1;
   984→  ctx.stroke();
   985→
   986→  // Arrow ready with better detail and volume
   987→  // The last arrow is a magnet arrow (magenta/purple) if the upgrade is purchased
   988→  if (!s.reloading && s.arrows > 0) {
   989→    const isLastMagnet = s.magnetArrow > 0 && s.arrows === 1;
   990→    const aR = isLastMagnet ? 255 : 0;
   991→    const aG = isLastMagnet ? 0 : 255;
   992→    const aB = isLastMagnet ? 255 : 255;
   993→    const arrowLen = 42 * pulse;
   994→
   995→    // Magnet aura around the nocked arrow
   996→    if (isLastMagnet) {
   997→      const pulseT = Date.now() / 150;
   998→      for (let ring = 0; ring < 2; ring++) {
   999→        const phase = (pulseT + ring * 1.3) % 2.6;
  1000→        const ringR = 14 + phase * 12;
  1001→        const ringAlpha = Math.max(0, 0.45 - phase / 5.5);
  1002→        ctx.beginPath();
  1003→        ctx.arc(arrowLen * 0.4, 0, ringR, 0, Math.PI * 2);
  1004→        ctx.strokeStyle = `rgba(255, 0, 255, ${ringAlpha})`;
  1005→        ctx.lineWidth = 2;
  1006→        ctx.stroke();
  1007→      }
  1008→    }
  1009→
  1010→    // Arrow shaft glow
  1011→    ctx.beginPath();
  1012→    ctx.moveTo(-15, 0);
  1013→    ctx.lineTo(arrowLen, 0);
  1014→    ctx.strokeStyle = `rgba(${aR}, ${aG}, ${aB}, ${isLastMagnet ? 0.3 : 0.2})`;
  1015→    ctx.lineWidth = isLastMagnet ? 14 : 12;
  1016→    ctx.stroke();
  1017→
  1018→    ctx.beginPath();
  1019→    ctx.moveTo(-15, 0);
  1020→    ctx.lineTo(arrowLen, 0);
  1021→    ctx.strokeStyle = `rgba(${aR}, ${aG}, ${aB}, ${isLastMagnet ? 0.75 : 0.6})`;
  1022→    ctx.lineWidth = isLastMagnet ? 5 : 4;
  1023→    ctx.stroke();
  1024→
  1025→    // Arrow shaft core
  1026→    ctx.beginPath();
  1027→    ctx.moveTo(-15, 0);
  1028→    ctx.lineTo(arrowLen, 0);
  1029→    ctx.strokeStyle = '#ffffff';
  1030→    ctx.lineWidth = 2;
  1031→    ctx.shadowBlur = 0;
  1032→    ctx.stroke();
  1033→
  1034→    // High-tech Arrow head (3D)
  1035→    ctx.beginPath();
  1036→    ctx.moveTo(arrowLen + 8, 0);
  1037→    ctx.lineTo(arrowLen - 6, -8);
  1038→    ctx.lineTo(arrowLen - 2, 0);
  1039→    ctx.lineTo(arrowLen - 6, 8);
  1040→    ctx.closePath();
  1041→    const headGrad = ctx.createLinearGradient(arrowLen - 6, -8, arrowLen + 8, 8);
  1042→    headGrad.addColorStop(0, '#ffffff');
  1043→    headGrad.addColorStop(0.4, isLastMagnet ? '#ff00ff' : '#00ffff');
  1044→    headGrad.addColorStop(1, isLastMagnet ? '#660066' : '#005555');
  1045→    ctx.fillStyle = headGrad;
  1046→    ctx.fill();
  1047→
  1048→    // Arrow fletching (Energy fins)
  1049→    ctx.beginPath();
  1050→    ctx.moveTo(-15, 0);
  1051→    ctx.lineTo(-25, -8);
  1052→    ctx.lineTo(-18, 0);
  1053→    ctx.lineTo(-25, 8);
  1054→    ctx.closePath();
  1055→    ctx.fillStyle = `rgba(${aR}, ${aG}, ${aB}, 0.9)`;
  1056→    ctx.fill();
  1057→
  1058→    // Extra energy rings around the arrow
  1059→    ctx.beginPath();
  1060→    ctx.ellipse(0, 0, 4, 12, 0, 0, Math.PI * 2);
  1061→    ctx.strokeStyle = `rgba(${aR}, ${aG}, ${aB}, ${isLastMagnet ? 0.9 : 0.8})`;
  1062→    ctx.lineWidth = isLastMagnet ? 2 : 1.5;
  1063→    ctx.stroke();
  1064→  }
  1065→
  1066→  ctx.restore();
  1067→}
