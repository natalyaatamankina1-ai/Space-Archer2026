    33→    33→/** Обводка полигональной формы врага (общая для рендера и запекания). */
    34→    34→function tracePath(ctx: CanvasRenderingContext2D, shape: string, r: number) {
    35→    35→  switch (shape) {
    36→    36→    case 'circle':
    37→    37→      ctx.arc(0, 0, r, 0, Math.PI * 2);
    38→    38→      break;
    39→    39→    case 'triangle':
    40→    40→      for (let i = 0; i < 3; i++) {
    41→    41→        const a = (i * Math.PI * 2) / 3 - Math.PI / 2;
    42→    42→        if (i === 0) ctx.moveTo(Math.cos(a) * r, Math.sin(a) * r);
    43→    43→        else ctx.lineTo(Math.cos(a) * r, Math.sin(a) * r);
    44→    44→      }
    45→    45→      ctx.closePath();
    46→    46→      break;
    47→    47→    case 'square':
    48→    48→      ctx.rect(-r * 0.8, -r * 0.8, r * 1.6, r * 1.6);
    49→    49→      break;
    50→    50→    case 'diamond':
    51→    51→      ctx.moveTo(0, -r);
    52→    52→      ctx.lineTo(r * 0.8, 0);
    53→    53→      ctx.lineTo(0, r);
    54→    54→      ctx.lineTo(-r * 0.8, 0);
    55→    55→      ctx.closePath();
    56→    56→      break;
    57→    57→    case 'hexagon':
    58→    58→      for (let i = 0; i < 6; i++) {
    59→    59→        const a = (i * Math.PI * 2) / 6;
    60→    60→        if (i === 0) ctx.moveTo(Math.cos(a) * r, Math.sin(a) * r);
    61→    61→        else ctx.lineTo(Math.cos(a) * r, Math.sin(a) * r);
    62→    62→      }
    63→    63→      ctx.closePath();
    64→    64→      break;
    65→    65→    case 'pentagon':
    66→    66→      for (let i = 0; i < 5; i++) {
    67→    67→        const a = (i * Math.PI * 2) / 5 - Math.PI / 2;
    68→    68→        if (i === 0) ctx.moveTo(Math.cos(a) * r, Math.sin(a) * r);
    69→    69→        else ctx.lineTo(Math.cos(a) * r, Math.sin(a) * r);
    70→    70→      }
    71→    71→      ctx.closePath();
    72→    72→      break;
    73→    73→    case 'star':
    74→    74→      for (let i = 0; i < 10; i++) {
    75→    75→        const a = (i * Math.PI) / 5 - Math.PI / 2;
    76→    76→        const rad = i % 2 === 0 ? r : r * 0.5;
    77→    77→        if (i === 0) ctx.moveTo(Math.cos(a) * rad, Math.sin(a) * rad);
    78→    78→        else ctx.lineTo(Math.cos(a) * rad, Math.sin(a) * rad);
    79→    79→      }
    80→    80→      ctx.closePath();
    81→    81→      break;
    82→    82→    default:
    83→    83→      ctx.arc(0, 0, r, 0, Math.PI * 2);
    84→    84→  }
    85→    85→}

    87→    87→/** Запечённое ядро врага: градиентная заливка + glow-обводка + белая обводка. */
    88→    88→function getShapeSprite(shape: string, color: string, coreRadius: number): HTMLCanvasElement {
    89→    89→  const key = shape + '|' + color + '|' + coreRadius.toFixed(1);
    90→    90→  const cached = shapeSpriteCache.get(key);
    91→    91→  if (cached) return cached;
    92→    92→
    93→    93→  const margin = 12; // запас под широкую glow-обводку (lineWidth 15)
    94→    94→  const size = Math.ceil(coreRadius * 2 + margin * 2);
    95→    95→  const cv = document.createElement('canvas');
    96→    96→  cv.width = size;
    97→    97→  cv.height = size;
    98→    98→  const c = cv.getContext('2d')!;
    99→    99→  c.translate(size / 2, size / 2);
   100→   100→
   101→   101→  // Форма + радиальный градиент (как в оригинале: белый → цвет)
   102→   102→  c.beginPath();
   103→   103→  tracePath(c, shape, coreRadius);
   104→   104→  const grad = c.createRadialGradient(0, 0, 0, 0, 0, coreRadius);
   105→   105→  grad.addColorStop(0, '#ffffff');
   106→   106→  grad.addColorStop(0.3, color);
   107→   107→  grad.addColorStop(1, color);
   108→   108→  c.fillStyle = grad;
   109→   109→  c.fill();
   110→   110→
   111→   111→  // Широкая мягкая обводка-глоу + тонкая белая (как в оригинале)
   112→   112→  c.strokeStyle = color + '44';
   113→   113→  c.lineWidth = 15;
   114→   114→  c.stroke();
   115→   115→  c.strokeStyle = '#ffffff';
   116→   116→  c.lineWidth = 1.5;
   117→   117→  c.stroke();
   118→   118→
   119→   119→  shapeSpriteCache.set(key, cv);
   120→   120→  return cv;
   121→   121→}

   390→   390→export function createEnemy(
   391→   391→  type: Omit<EnemyType, 'hp'>,
   392→   392→  canvasW: number,
   393→   393→  canvasH: number,
   394→   394→  overrides?: Partial<Enemy>
   395→   395→): Enemy {
   396→   396→  if (!type) {
   397→   397→    type = ENEMY_TYPES.find(t => t.id === 'scout')!;
   398→   398→  }
   399→   399→  const side = Math.floor(Math.random() * 4);
   400→   400→  let spawnX = 0, spawnY = 0;
   401→   401→  const margin = 60;
   402→   402→  switch (side) {
   403→   403→    case 0: spawnX = Math.random() * canvasW; spawnY = -margin; break;
   404→   404→    case 1: spawnX = canvasW + margin; spawnY = Math.random() * canvasH; break;
   405→   405→    case 2: spawnX = Math.random() * canvasW; spawnY = canvasH + margin; break;
   406→   406→    case 3: spawnX = -margin; spawnY = Math.random() * canvasH; break;
   407→   407→  }
   408→   408→
   409→   409→  const finalX = overrides?.x ?? spawnX;
   410→   410→  const finalY = overrides?.y ?? spawnY;
   411→   411→  const speed = type.speed * (0.85 + Math.random() * 0.3);
   412→   412→  
   413→   413→  let angle;
   414→   414→  if (type.id === 'bouncer' && !overrides) {
   415→   415→    // Ricochet should aim at a wall, not the player
   416→   416→    const targetSide = (side + 1 + Math.floor(Math.random() * 3)) % 4;
   417→   417→    let tx = 0, ty = 0;
   418→   418→    switch (targetSide) {
   419→   419→      case 0: tx = Math.random() * canvasW; ty = 0; break;
   420→   420→      case 1: tx = canvasW; ty = Math.random() * canvasH; break;
   421→   421→      case 2: tx = Math.random() * canvasW; ty = canvasH; break;
   422→   422→      case 3: tx = 0; ty = Math.random() * canvasH; break;
   423→   423→    }
   424→   424→    angle = Math.atan2(ty - finalY, tx - finalX);
   425→   425→  } else {
   426→   426→    // Default: aim at player (center)
   427→   427→    angle = Math.atan2(canvasH / 2 - finalY, canvasW / 2 - finalX);
   428→   428→  }
   429→   429→
   430→   430→  return {
   431→   431→    ...type,
   432→   432→    hp: type.maxHp,
   433→   433→    x: finalX,
   434→   434→    y: finalY,
   435→   435→    vx: Math.cos(angle) * speed,
   436→   436→    vy: Math.sin(angle) * speed,
   437→   437→    angle,
   438→   438→    trail: [],
   439→   439→    zigzagTimer: 0,
   440→   440→    teleportTimer: type.id === 'phantom' ? Math.random() * 60 : Math.random() * 180,
   441→   441→    splitDone: false,
   442→   442→    pulseRadius: type.radius,
   443→   443→    pulseGrowing: true,
   444→   444→    flashTimer: 0,
   445→   445→    bouncerEntered: false,
   446→   446→    teleportAlpha: (type.pattern === 'teleport' || type.pattern === 'stealth') ? 0 : 1,
