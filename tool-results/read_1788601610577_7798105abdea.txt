   442→/** Smooth Catmull-Rom spline through `pts`, converted to a cubic Bézier `d` string. */
   443→function catmullRomPath(pts: { x: number; y: number }[]): string {
   444→  if (pts.length < 2) return '';
   445→  if (pts.length === 2) {
   446→    return `M ${pts[0].x.toFixed(1)} ${pts[0].y.toFixed(1)} L ${pts[1].x.toFixed(1)} ${pts[1].y.toFixed(1)}`;
   447→  }
   448→  let d = `M ${pts[0].x.toFixed(1)} ${pts[0].y.toFixed(1)}`;
   449→  for (let i = 0; i < pts.length - 1; i++) {
   450→    const p0 = pts[Math.max(0, i - 1)];
   451→    const p1 = pts[i];
   452→    const p2 = pts[i + 1];
   453→    const p3 = pts[Math.min(pts.length - 1, i + 2)];
   454→    const c1x = p1.x + (p2.x - p0.x) / 6;
   455→    const c1y = p1.y + (p2.y - p0.y) / 6;
   456→    const c2x = p2.x - (p3.x - p1.x) / 6;
   457→    const c2y = p2.y - (p3.y - p1.y) / 6;
   458→    d += ` C ${c1x.toFixed(1)} ${c1y.toFixed(1)}, ${c2x.toFixed(1)} ${c2y.toFixed(1)}, ${p2.x.toFixed(1)} ${p2.y.toFixed(1)}`;
   459→  }
   460→  return d;
   461→}

   464→ * One curved connection link, rendered as SVG and mirroring the layered look
   465→ * of the straight div links 1:1 (wide glow → two-color base → flowing pulse
   466→ * → white core → traveling orb → breathing sockets), including the locked
   467→ * dashed-shimmer variant.
   468→ */
   469→function buildCurvedLink(
   470→  linkKey: string,
   471→  from: { x: number; y: number; color: string; unlocked: boolean },
   472→  to: { x: number; y: number; color: string; unlocked: boolean },
   473→  item: UpgradeItem,
   474→  waypoints: { x: number; y: number }[],
   475→): React.ReactElement {
   476→  // Regular node ring radius (the base node never uses curved links)
   477→  const NODE_R = 48;
   478→  const w0 = waypoints[0];
   479→  const wLast = waypoints[waypoints.length - 1];
   480→  const d0 = Math.hypot(w0.x - from.x, w0.y - from.y) || 1;
   481→  const sx = from.x + ((w0.x - from.x) / d0) * NODE_R;
   482→  const sy = from.y + ((w0.y - from.y) / d0) * NODE_R;
   483→  const dl = Math.hypot(wLast.x - to.x, wLast.y - to.y) || 1;
   484→  const ex = to.x + ((wLast.x - to.x) / dl) * NODE_R;
   485→  const ey = to.y + ((wLast.y - to.y) / dl) * NODE_R;
   486→  const d = catmullRomPath([{ x: sx, y: sy }, ...waypoints, { x: ex, y: ey }]);
   487→  const isUnlocked = from.unlocked;
   488→  const childUnlocked = to.unlocked;
   489→  const gid = `cl-${linkKey}`;
   490→  const orbDur = `${2.4 + (item.key.length % 3) * 0.4}s`;
   491→  const orbBegin = `${(item.key.charCodeAt(0) % 10) * 0.2}s`;
   492→
   493→  return (
   494→    <g key={linkKey}>
   495→      <defs>
   496→        <linearGradient id={`${gid}-glow`} gradientUnits="userSpaceOnUse" x1={sx} y1={sy} x2={ex} y2={ey}>
   497→          <stop offset="0%" stopColor={from.color} stopOpacity={0} />
   498→          <stop offset="15%" stopColor={from.color} stopOpacity={0.33} />
   499→          <stop offset="85%" stopColor={to.color} stopOpacity={0.33} />
   500→          <stop offset="100%" stopColor={to.color} stopOpacity={0} />
   501→        </linearGradient>
   502→        <linearGradient id={`${gid}-base`} gradientUnits="userSpaceOnUse" x1={sx} y1={sy} x2={ex} y2={ey}>
   503→          <stop offset="0%" stopColor={from.color} />
   504→          <stop offset="100%" stopColor={to.color} />
   505→        </linearGradient>
   506→        <linearGradient id={`${gid}-core`} gradientUnits="userSpaceOnUse" x1={sx} y1={sy} x2={ex} y2={ey}>
   507→          <stop offset="0%" stopColor="#ffffff" stopOpacity={0.3} />
   508→          <stop offset="50%" stopColor="#ffffff" stopOpacity={0.95} />
   509→          <stop offset="100%" stopColor="#ffffff" stopOpacity={0.3} />
   510→        </linearGradient>
   511→        <radialGradient id={`${gid}-orb`}>
   512→          <stop offset="0%" stopColor="#ffffff" />
   513→          <stop offset="40%" stopColor={to.color} />
   514→          <stop offset="70%" stopColor={to.color} stopOpacity={0} />
   515→        </radialGradient>
   516→        <radialGradient id={`${gid}-sock-from`}>
   517→          <stop offset="0%" stopColor="#ffffff" />
   518→          <stop offset="60%" stopColor={from.color} />
   519→        </radialGradient>
   520→        <radialGradient id={`${gid}-sock-to`}>
   521→          <stop offset="0%" stopColor="#ffffff" />
   522→          <stop offset="60%" stopColor={to.color} />
   523→        </radialGradient>
   524→        <radialGradient id={`${gid}-sock-to-dim`}>
   525→          <stop offset="0%" stopColor={to.color} />
   526→          <stop offset="60%" stopColor={to.color} stopOpacity={0.53} />
   527→          <stop offset="100%" stopColor={to.color} stopOpacity={0} />
   528→        </radialGradient>
   529→      </defs>
   530→      {isUnlocked ? (
   531→        <>
   532→          {/* Layer 1 — wide soft outer glow */}
   533→          <path d={d} fill="none" stroke={`url(#${gid}-glow)`} strokeWidth={16} strokeLinecap="round"
   534→            opacity={childUnlocked ? 0.9 : 0.55} style={{ filter: 'blur(7px)' }} />
   535→          {/* Layer 2 — steady two-color base gradient */}
   536→          <path d={d} fill="none" stroke={`url(#${gid}-base)`} strokeWidth={5} strokeLinecap="round"
   537→            opacity={0.75} style={{ filter: `drop-shadow(0 0 8px ${from.color}88)` }} />
   538→          {/* Layer 3 — flowing multi-pulse energy (one 30%-wide pulse per
   539→              125% period, sliding parent → child like tree-link-flow) */}
   540→          <path d={d} fill="none" stroke={`url(#${gid}-base)`} strokeWidth={3} strokeLinecap="round"
   541→            pathLength={100} strokeDasharray="30 95" opacity={0.95} style={{ mixBlendMode: 'screen' }}>
   542→            <animate attributeName="stroke-dashoffset" from="0" to="-125" dur="2.4s" repeatCount="indefinite" />
   543→          </path>
   544→          {/* Layer 4 — bright white core thread */}
   545→          <path d={d} fill="none" stroke={`url(#${gid}-core)`} strokeWidth={1.5} strokeLinecap="round"
   546→            style={{ filter: 'drop-shadow(0 0 4px rgba(255,255,255,0.6))' }} />
   547→          {/* Layer 5 — traveling energy orb (fades/scales at both ends,
   548→              exactly like tree-link-pulse-travel) */}
   549→          <circle r={7} fill={`url(#${gid}-orb)`} opacity={0}
   550→            style={{ filter: `blur(0.5px) drop-shadow(0 0 12px ${to.color})` }}>
   551→            <animateMotion dur={orbDur} begin={orbBegin} repeatCount="indefinite" path={d} />
   552→            <animate attributeName="opacity" values="0;1;1;0" keyTimes="0;0.1;0.9;1" dur={orbDur} begin={orbBegin} repeatCount="indefinite" />
   553→            <animate attributeName="r" values="2.8;7;7;2.8" keyTimes="0;0.1;0.9;1" dur={orbDur} begin={orbBegin} repeatCount="indefinite" />
   554→          </circle>
   555→          {/* Start socket */}
   556→          <circle cx={sx} cy={sy} r={6} fill={`url(#${gid}-sock-from)`}
   557→            style={{ filter: `drop-shadow(0 0 10px ${from.color}) drop-shadow(0 0 20px ${from.color}88)`, animation: 'tree-socket-breathe 2.5s ease-in-out infinite' }} />
   558→          {/* End socket */}
   559→          <circle cx={ex} cy={ey} r={childUnlocked ? 6 : 4.5}
   560→            fill={childUnlocked ? `url(#${gid}-sock-to)` : `url(#${gid}-sock-to-dim)`}
   561→            stroke={childUnlocked ? 'none' : `${to.color}aa`} strokeWidth={1}
   562→            style={{ filter: `drop-shadow(0 0 ${childUnlocked ? 10 : 6}px ${to.color}${childUnlocked ? '88' : '66'})`, animation: 'tree-socket-breathe 2.5s ease-in-out infinite' }} />
   563→        </>
   564→      ) : (
   565→        <>
   566→          {/* Locked link — dim dashed shimmer */}
   567→          <path d={d} fill="none" stroke="rgba(255,255,255,0.18)" strokeWidth={6} strokeLinecap="round"
   568→            strokeDasharray="14 18" opacity={0.5} style={{ filter: 'blur(0.5px)' }}>
   569→            <animate attributeName="stroke-dashoffset" from="0" to="-32" dur="2s" repeatCount="indefinite" />
   570→          </path>
   571→          <circle cx={sx} cy={sy} r={4} fill="rgba(0,0,0,0.6)" stroke="rgba(255,255,255,0.2)" strokeWidth={1} />
   572→          <circle cx={ex} cy={ey} r={4} fill="rgba(0,0,0,0.6)" stroke="rgba(255,255,255,0.12)" strokeWidth={1} />
   573→        </>
   574→      )}
   575→    </g>
   576→  );
   577→}

   937→/**
   938→ * Pan + wheel-zoom canvas for the upgrade tree.
   939→ * Defined at module scope (NOT inside UpgradesScreen) so it keeps a stable
   940→ * component identity across re-renders — otherwise its refs/transform would
   941→ * reset every time the parent updates state.
   942→ */
   943→function UpgradeTreeCanvas({
   944→  children,
   945→  onZoomChange,
   946→}: {
   947→  children: React.ReactNode;
   948→  onZoomChange: (zoom: number) => void;
   949→}) {
   950→  const { t } = useLanguage();
   951→  const containerRef = useRef<HTMLDivElement>(null);
   952→  const contentRef = useRef<HTMLDivElement>(null);
   953→  const transformRef = useRef({ zoom: 1, panX: 0, panY: 0 });
   954→  // displayZoom mirrors the ref into state so we can read it during render
   955→  // for the zoom indicator (refs must not be read during render).
   956→  const [displayZoom, setDisplayZoom] = useState(1);
   957→  const [isDragging, setIsDragging] = useState(false);
   958→  const dragStart = useRef({ x: 0, y: 0, panX: 0, panY: 0, moved: false });
   959→
   960→  const applyTransform = useCallback(() => {
   961→    const el = contentRef.current;
   962→    if (el) {
   963→      const { zoom, panX, panY } = transformRef.current;
   964→      el.style.transform = `translate(${panX}px, ${panY}px) scale(${zoom})`;
   965→    }
   966→    onZoomChange(transformRef.current.zoom);
   967→    setDisplayZoom(transformRef.current.zoom);
   968→  }, [onZoomChange]);
   969→
   970→  // Center the content on first mount
   971→  useEffect(() => {
   972→    const el = containerRef.current;
   973→    if (!el) return;
   974→    const cw = el.clientWidth;
   975→    const ch = el.clientHeight;
   976→    transformRef.current = {
   977→      zoom: 1,
   978→      panX: cw / 2 - CENTER,
   979→      panY: ch / 2 - CENTER,
   980→    };
   981→    applyTransform();
   982→    const handleResize = () => {
   983→      const z = transformRef.current.zoom;
   984→      transformRef.current.panX = el.clientWidth / 2 - CENTER * z;
   985→      transformRef.current.panY = el.clientHeight / 2 - CENTER * z;
   986→      applyTransform();
   987→    };
   988→    window.addEventListener('resize', handleResize);
   989→    return () => window.removeEventListener('resize', handleResize);
   990→  }, [applyTransform]);
   991→
   992→  // Non-passive wheel listener so we can preventDefault page scroll
   993→  useEffect(() => {
   994→    const el = containerRef.current;
   995→    if (!el) return;
   996→    const onWheel = (e: WheelEvent) => {
   997→      e.preventDefault();
   998→      const rect = el.getBoundingClientRect();
   999→      const cx = e.clientX - rect.left;
  1000→      const cy = e.clientY - rect.top;
  1001→      const { zoom: prevZoom, panX: prevPanX, panY: prevPanY } = transformRef.current;
  1002→      const factor = e.deltaY < 0 ? 1.12 : 1 / 1.12;
  1003→      const newZoom = clampZoom(prevZoom * factor);
  1004→      const actualFactor = newZoom / prevZoom;
  1005→      transformRef.current.zoom = newZoom;
  1006→      transformRef.current.panX = cx - (cx - prevPanX) * actualFactor;
  1007→      transformRef.current.panY = cy - (cy - prevPanY) * actualFactor;
  1008→      applyTransform();
  1009→    };
  1010→    el.addEventListener('wheel', onWheel, { passive: false });
  1011→    return () => el.removeEventListener('wheel', onWheel);
  1012→  }, [applyTransform]);
  1013→
  1014→  const onMouseDown = (e: React.MouseEvent) => {
  1015→    if (e.button !== 0) return;
  1016→    dragStart.current = {
  1017→      x: e.clientX,
  1018→      y: e.clientY,
  1019→      panX: transformRef.current.panX,
  1020→      panY: transformRef.current.panY,
  1021→      moved: false,
  1022→    };
  1023→    setIsDragging(true);
  1024→  };
  1025→
  1026→  const onMouseMove = (e: React.MouseEvent) => {
  1027→    if (!isDragging) return;
  1028→    const dx = e.clientX - dragStart.current.x;
  1029→    const dy = e.clientY - dragStart.current.y;
  1030→    if (Math.abs(dx) > 3 || Math.abs(dy) > 3) dragStart.current.moved = true;
  1031→    transformRef.current.panX = dragStart.current.panX + dx;
  1032→    transformRef.current.panY = dragStart.current.panY + dy;
  1033→    applyTransform();
  1034→  };
  1035→
  1036→  const zoomBy = useCallback((factor: number) => {
  1037→    const el = containerRef.current;
  1038→    if (!el) return;
  1039→    const cx = el.clientWidth / 2;
  1040→    const cy = el.clientHeight / 2;
  1041→    const { zoom: prevZoom, panX: prevPanX, panY: prevPanY } = transformRef.current;
  1042→    const newZoom = clampZoom(prevZoom * factor);
  1043→    const actualFactor = newZoom / prevZoom;
  1044→    transformRef.current.zoom = newZoom;
  1045→    transformRef.current.panX = cx - (cx - prevPanX) * actualFactor;
  1046→    transformRef.current.panY = cy - (cy - prevPanY) * actualFactor;
  1047→    applyTransform();
  1048→  }, [applyTransform]);
  1049→
  1050→  const resetView = useCallback(() => {
  1051→    const el = containerRef.current;
  1052→    if (!el) return;
  1053→    transformRef.current = {
  1054→      zoom: 1,
  1055→      panX: el.clientWidth / 2 - CENTER,
  1056→      panY: el.clientHeight / 2 - CENTER,
  1057→    };
  1058→    applyTransform();
  1059→  }, [applyTransform]);
  1060→
  1061→  const currentZoom = displayZoom;
  1062→
  1063→  return (
  1064→    <div
  1065→      className="w-full h-[65vh] min-h-[500px] overflow-hidden rounded-3xl border border-white/15 shadow-2xl relative bg-black/40 touch-none select-none"
  1066→      ref={containerRef}
  1067→      onMouseDown={onMouseDown}
  1068→      onMouseUp={() => setIsDragging(false)}
  1069→      onMouseLeave={() => setIsDragging(false)}
  1070→      onMouseMove={onMouseMove}
  1071→      style={{ cursor: isDragging ? 'grabbing' : 'grab' }}
  1072→    >
  1073→      <div
  1074→        ref={contentRef}
  1075→        className="absolute top-0 left-0"
  1076→        style={{
  1077→          width: CONTENT_SIZE,
  1078→          height: CONTENT_SIZE,
  1079→          transformOrigin: '0 0',
  1080→          willChange: 'transform',
  1081→        }}
  1082→      >
  1083→        <div className="absolute inset-0 grid-bg opacity-30 pointer-events-none" />
  1084→        <div className="absolute inset-0 bg-[radial-gradient(ellipse_at_center,rgba(0,255,255,0.06),transparent_60%)] pointer-events-none" />
  1085→        {children}
  1086→      </div>
  1087→
  1088→      {/* Zoom controls overlay */}
  1089→      <div className="absolute bottom-3 right-3 flex flex-col gap-1.5 z-40">
  1090→        <button
  1091→          onClick={() => { audio.playClick(); zoomBy(1.2); }}
  1092→          className="w-9 h-9 rounded-lg glass-panel border border-cyan-500/30 text-cyan-300 font-orbitron font-bold text-lg flex items-center justify-center hover:border-cyan-400/60 hover:bg-cyan-500/10 transition-all shadow-[0_0_10px_rgba(0,255,255,0.15)]"
  1093→          title={t('upgrades.zoomIn')}
  1094→        >
  1095→          +
  1096→        </button>
  1097→        <button
  1098→          onClick={() => { audio.playClick(); zoomBy(1 / 1.2); }}
  1099→          className="w-9 h-9 rounded-lg glass-panel border border-cyan-500/30 text-cyan-300 font-orbitron font-bold text-lg flex items-center justify-center hover:border-cyan-400/60 hover:bg-cyan-500/10 transition-all shadow-[0_0_10px_rgba(0,255,255,0.15)]"
  1100→          title={t('upgrades.zoomOut')}
  1101→        >
  1102→          −
  1103→        </button>
  1104→        <button
  1105→          onClick={() => { audio.playClick(); resetView(); }}
  1106→          className="w-9 h-9 rounded-lg glass-panel border border-cyan-500/30 text-cyan-300 font-orbitron font-bold text-[10px] flex items-center justify-center hover:border-cyan-400/60 hover:bg-cyan-500/10 transition-all shadow-[0_0_10px_rgba(0,255,255,0.15)]"
  1107→          title={t('upgrades.zoomReset')}
  1108→        >
  1109→          1:1
  1110→        </button>
  1111→      </div>
  1112→
  1113→      {/* Zoom level indicator */}
  1114→      <div className="absolute bottom-3 left-3 z-40 glass-panel px-3 py-1.5 rounded-lg border border-white/10 font-orbitron text-xs text-cyan-300/80 tracking-widest shadow-[0_0_10px_rgba(0,0,0,0.4)]">
  1115→        {Math.round(currentZoom * 100)}%
  1116→      </div>
  1117→    </div>
